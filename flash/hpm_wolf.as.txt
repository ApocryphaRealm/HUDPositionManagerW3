
      // ---------------------------------------------------------------------------------------------------------------
      // HUD Position Manager: straight bars (the owner, 2026-10-07): "the stamina is a straight bar going in the same
      // direction as the health right underneath it, and the same for the adrenaline underneath stamina", "Toxicity
      // will be beneath the adrenaline bar", then: "all the bars except for health to be a lot smaller than the health
      // bar ... using the original bar's width or thickness, but in a straight line" and "use the original bar's length
      // as well, so that each bar below health is slightly shorter than the bar above it", and no colour or outline
      // outside the bar (the first try copied the health bar's art, whose frame pieces showed).
      // So each bar is drawn here: a dark track, a two-tone fill cut to the value, a thin border, the toxicity bar's
      // pointed end. Thickness = the game's own straight toxicity bar; lengths = the game's three bars (stamina arc,
      // adrenaline arc, toxicity bar) measured along their art, longest first, each at least a little shorter than the
      // one above. They read the values the game sends its own bars, which are faded out (alpha: the game sets their
      // visibility itself, e.g. for Ciri). Adrenaline carries marks at its thirds - the three points it fills. The
      // skull badge sits at the toxicity bar's left end as in the game, the sign badge beside it.
      // ---------------------------------------------------------------------------------------------------------------
      private static const HPM_COLOURS:Array = [0xE8B83A,0x4E9BDA,0x62B83E,0x8A4FE0];   // stamina, adrenaline, toxicity, magic (arcane violet-blue)
      private var hpmFills:Array = [];      // the fill's mask per bar (scaleX = value)
      private var hpmRows:Array = [];       // each bar's sprite
      private var hpmValues:Array = [0,0,0,0];

      private function hpmShade(c:uint, k:Number) : uint
      {
         var r:int = Math.min(255,(c >> 16 & 255) * k);
         var g:int = Math.min(255,(c >> 8 & 255) * k);
         var b:int = Math.min(255,(c & 255) * k);
         return r << 16 | g << 8 | b;
      }

      // the bar's outline: a rectangle whose right end comes to a point (the toxicity bar's shape)
      private function hpmShape(gfx:*, w:Number, h:Number) : void
      {
         var tip:Number = Math.min(h * 0.55,w * 0.3);
         gfx.moveTo(0,0);
         gfx.lineTo(w - tip,0);
         gfx.lineTo(w,h / 2);
         gfx.lineTo(w - tip,h);
         gfx.lineTo(0,h);
         gfx.lineTo(0,0);
      }

      private function hpmMakeBar(colour:uint, w:Number, h:Number, thirds:Boolean) : Sprite
      {
         var row:Sprite = new Sprite();
         var track:Shape = new Shape();
         var fill:Shape = new Shape();
         var cut:Shape = new Shape();
         var edge:Shape = new Shape();
         var m:Matrix = new Matrix();
         var i:int = 0;
         track.graphics.beginFill(0x080808,0.88);
         this.hpmShape(track.graphics,w,h);
         track.graphics.endFill();
         m.createGradientBox(w,h,Math.PI / 2,0,0);
         fill.graphics.beginGradientFill(GradientType.LINEAR,[this.hpmShade(colour,1.25),colour,this.hpmShade(colour,0.62)],[1,1,1],[0,110,255],m);
         this.hpmShape(fill.graphics,w,h);
         fill.graphics.endFill();
         cut.graphics.beginFill(0);
         cut.graphics.drawRect(0,-1,w,h + 2);
         cut.graphics.endFill();
         fill.mask = cut;
         edge.graphics.lineStyle(1,0x2A2620,1,true);
         this.hpmShape(edge.graphics,w,h);
         row.addChild(track);
         row.addChild(fill);
         row.addChild(cut);
         row.addChild(edge);
         if(thirds)
         {
            // the dividers are a shape of their own (hpmFocusTicks), so Witcher Gear Overhaul's Magic can hide them
            this.hpmFocusTicks = new Shape();
            this.hpmFocusW = w;
            this.hpmFocusH = h;
            row.addChild(this.hpmFocusTicks);
            this.hpmDrawTicks();
         }
         this.hpmFills.push(cut);
         return row;
      }

      // ---------------------------------------------------------------------------------------------------------------
      // The adrenaline bar as one continuous bar (Magic, Witcher Gear Overhaul): its Magic design draws the pool as a
      // single bar, so the dividers can be switched off and the blink at each third skipped. With nothing called, 3
      // segments and a maximum of 3 - the game's own look.
      // ---------------------------------------------------------------------------------------------------------------
      private var hpmFocusTicks:Shape = null;
      private var hpmFocusW:Number = 0;
      private var hpmFocusH:Number = 0;
      private var hpmFocusSegs:int = 3;
      private var hpmFocusMax:Number = 3;
      private var hpmFocusRaw:Number = 0;
      private var hpmFocusTf:TextField = null;
      private var hpmFocusNumOn:Boolean = false;

      private function hpmDrawTicks() : void
      {
         var i:int = 1;
         if(!this.hpmFocusTicks)
         {
            return;
         }
         this.hpmFocusTicks.graphics.clear();
         this.hpmFocusTicks.visible = this.hpmFocusSegs >= 2;
         if(this.hpmFocusSegs < 2)
         {
            return;
         }
         this.hpmFocusTicks.graphics.lineStyle(1.5,0xC8C8C8,0.7);
         while(i < this.hpmFocusSegs)
         {
            this.hpmFocusTicks.graphics.moveTo(this.hpmFocusW * i / this.hpmFocusSegs,1.5);
            this.hpmFocusTicks.graphics.lineTo(this.hpmFocusW * i / this.hpmFocusSegs,this.hpmFocusH - 1.5);
            i++;
         }
      }

      // n <= 1: no dividers, one continuous bar and no blink at a third; n >= 2: n - 1 evenly spaced marks (3 = the game's)
      public function hpmSetFocusSegments(n:int) : void
      {
         this.hpmFocusSegs = n;
         this.hpmDrawTicks();
      }

      // the adrenaline value that means "full" (the game's is 3)
      public function setFocusMax(m:Number) : void
      {
         if(m > 0)
         {
            this.hpmFocusMax = m;
            this.hpmSet(1,this.hpmFocusRaw * 100 / m);
         }
      }

      // the game asked for the whole-number change blink: not in single-segment mode, except when the bar becomes full
      public function hpmBlinkOk(points:int) : Boolean
      {
         return this.hpmFocusSegs > 1 || points >= this.hpmFocusMax;
      }

      // the game's adrenaline value (0..hpmFocusMax) -> the bar
      public function hpmFocusValue(v:Number) : void
      {
         this.hpmFocusRaw = v;
         this.hpmSet(1,v * 100 / this.hpmFocusMax);
      }

      // a small "current / max" number at the bar's right end: show it (true) with the two numbers, or hide it (false)
      public function hpmSetFocusNumber(cur:Number, max:Number, show:Boolean) : void
      {
         var fmt:TextFormat = null;
         this.hpmFocusNumOn = show;
         if(show && !this.hpmFocusTf)
         {
            this.hpmFocusTf = new TextField();
            fmt = new TextFormat("$NormalFont",Math.max(10,this.hpmFocusH * 1.15),0xFFFFFF);
            this.hpmFocusTf.defaultTextFormat = fmt;
            this.hpmFocusTf.selectable = false;
            this.hpmFocusTf.mouseEnabled = false;
            this.hpmFocusTf.autoSize = "left";
            this.hpmFocusTf.wordWrap = false;
            this.hpmFocusTf.multiline = false;
            addChild(this.hpmFocusTf);
         }
         if(this.hpmFocusTf)
         {
            this.hpmFocusTf.text = Math.round(cur) + " / " + Math.round(max);
            this.hpmApplyParts();
         }
      }

      // the number sits just past the adrenaline bar, moved and sized with it
      private function hpmPlaceNumber() : void
      {
         var row:DisplayObject = null;
         var b:Rectangle = null;
         if(!this.hpmFocusTf || this.hpmRows.length < 2)
         {
            return;
         }
         row = DisplayObject(this.hpmRows[1]);
         b = row.getBounds(this);
         this.hpmFocusTf.scaleX = Math.max(0.1,Math.abs(row.scaleY));
         this.hpmFocusTf.scaleY = Math.max(0.1,Math.abs(row.scaleY));
         this.hpmFocusTf.x = b.right + 4;
         this.hpmFocusTf.y = b.top + (b.height - this.hpmFocusTf.height) / 2;
         this.hpmFocusTf.visible = this.hpmFocusNumOn && row.visible;
      }

      // ---------------------------------------------------------------------------------------------------------------
      // A Magic bar of its own (Witcher Gear Overhaul: Magic is a separate pool that only Signs spend). The fourth bar,
      // under the toxicity bar, drawn like the others in an arcane colour. Hidden until a mod calls hpmShowMagic(true), so
      // without such a mod the widget is exactly as before. It is the part "Magic" of the layout: moved, sized, hidden and
      // shown like the other bars. hpmSetMagic(cur, max) sets the fill (cur / max) and the optional "cur / max" number.
      // ---------------------------------------------------------------------------------------------------------------
      private var hpmMagicOn:Boolean = false;
      private var hpmMagicNumOn:Boolean = false;
      private var hpmMagicTf:TextField = null;
      private var hpmMagicCur:Number = 0;
      private var hpmMagicMax:Number = 0;

      // the fill and the number: cur of max (max <= 0 = empty)
      public function hpmSetMagic(cur:Number, max:Number) : void
      {
         this.hpmMagicCur = cur;
         this.hpmMagicMax = max;
         this.hpmSet(3,max > 0 ? cur * 100 / max : 0);
         if(this.hpmMagicTf)
         {
            this.hpmMagicTf.text = Math.round(cur) + " / " + Math.round(max);
         }
         this.hpmPlaceMagicNumber();
      }

            return;

      // length along an object's art: a straight bar's width; an arc's chord stretched by a quarter-circle's arc/chord
      private function hpmLength(d:DisplayObject, straight:Boolean) : Number
      {
         var b:Rectangle = null;
         if(!d)
         {
            return 0;
         }
         b = d.getBounds(this);
         if(straight)
         {
            return b.width;
         }
         return Math.sqrt(b.width * b.width + b.height * b.height) * 1.11;
      }

      public function hpmBarsInit() : void
      {
         var fill:Rectangle = null;
         var med:Rectangle = null;
         var left:Number = NaN;
         var t:Number = NaN;
         var gap:Number = NaN;
         var lens:Array = null;
         var row:Sprite = null;
         var i:int = 0;
         var tox:Rectangle = null;
         var y:Number = NaN;
         var maxLen:Number = NaN;
         var xpAt:int = -1;
         var xpLine:Rectangle = null;
         if(!this.mcHealthBar || this.hpmRows.length > 0)
         {
            return;
         }
         fill = this.hpmFillBox(this.mcHealthBar);
         med = !this.mcWolfsHead ? null : this.mcWolfsHead.getBounds(this);
         // where the health bar shows: right of the medallion (it covers the bar's left end)
         left = !med ? fill.left : Math.max(fill.left,med.right - 6);
         // thickness: the game's own straight toxicity bar (about half the health bar)
         tox = !this.mcToxicityBar ? null : this.mcToxicityBar.getBounds(this);
         t = !tox || tox.height <= 0 ? fill.height * 0.5 : Math.min(tox.height,fill.height * 0.55);
         gap = Math.max(2,t * 0.55);
         lens = [this.hpmLength(this.mcToxicityBar,true),this.hpmLength(this.mcAdrenalinePoints,false),this.hpmLength(this.mcStaminaBar,false)];
         lens.sort(Array.NUMERIC | Array.DESCENDING);
         maxLen = fill.right - left;
         i = 0;
         while(i < 3)
         {
            if(!(lens[i] > t * 3))
            {
               lens[i] = maxLen * (0.62 - 0.08 * i);
            }
            lens[i] = Math.min(lens[i],(i == 0 ? maxLen : lens[i - 1]) * 0.9);
            i++;
         }
         y = fill.bottom + gap * 1.6;
         // the experience (level) line above the health bar, lined up with it as the bars below are: from the same left
         // edge to the health bar's right end, as far above it as the stamina bar is below it (hpmPlaceXp)
         this.hpmPlaceXp(left,fill.right,fill.top - gap * 1.6);
         // the badges (the owner, 2026-10-07): the sign badge to the LEFT of the adrenaline bar, where the game has it;
         // the skull badge at the toxicity bar's left end, and that bar starts just right of it - its right end stays
         // where its length puts it, so it is still a little shorter than the bar above. The two badges sit on
         // neighbouring rows but side by side (sign left of the bars' start, skull right of it), so they never overlap.
         this.hpmPlaceBadge(this.mcSignSlot,[this.mcSignSlot,this.mSignReady,this.mcSignText],left - 3,y + (t + gap) + t / 2,1);
         this.hpmPlaceBadge(this.mcSkullBck || this.mcSkull,[this.mcSkullBck,this.mcSkull],left,y + (t + gap) * 2 + t / 2,2);
         tox = !(this.mcSkullBck || this.mcSkull) ? null : DisplayObject(this.mcSkullBck || this.mcSkull).getBounds(this);
         i = 0;
         while(i < 3)
         {
            row = null;
            if(i == 2 && tox && tox.right + 2 < left + lens[i] - t * 3)
            {
               row = this.hpmMakeBar(HPM_COLOURS[i],left + lens[i] - (tox.right + 2),t,false);
               row.x = tox.right + 2;
            }
            else
            {
               row = this.hpmMakeBar(HPM_COLOURS[i],lens[i],t,i == 1);
               row.x = left;
            }
            row.y = y + (t + gap) * i;
            addChildAt(row,getChildIndex(this.mcHealthBar));
            this.hpmRows.push(row);
            i++;
         }
         // the Magic bar (hidden until hpmShowMagic): under the toxicity bar, 0.9 of its length
         row = this.hpmMakeBar(HPM_COLOURS[3],lens[2] * 0.9,t,false);
         row.x = left;
         row.y = y + (t + gap) * 3;
         row.visible = false;
         addChildAt(row,getChildIndex(this.mcHealthBar));
         this.hpmRows.push(row);
         this.hpmHideRound();
         this.hpmRowBase = [];
         i = 0;
         while(i < this.hpmRows.length)
         {
            this.hpmRowBase.push([DisplayObject(this.hpmRows[i]).x,DisplayObject(this.hpmRows[i]).y]);
            i++;
         }
         // the game's own pieces of each other part, each in a container of its own (moved, sized and shown as one)
         this.hpmWrap(3,[this.mcSignSlot,this.mSignReady,this.mcSignText]);
         this.hpmWrap(4,[this.mcSkullBck,this.mcSkull]);
         this.hpmWrap(5,[this.mcBckCircle,this.mcWolfsHead,this.mcMutationFeedback]);
         this.hpmWrap(6,[this.mcHealthBar]);
         // the experience bar with its level-up mark (the diamond at the line's right end); the container takes the line's
         // own depth, under the medallion, and is measured by the line, not by the bar's value mask (hpmLineBox)
         xpAt = !this.mcExperienceBar || this.mcExperienceBar.parent != this ? -1 : getChildIndex(this.mcExperienceBar);
         this.hpmWrap(7,[this.mcExperienceBar,this.mcNewLevelIndcator]);
         if(this.hpmBoxes[7] && xpAt >= 0)
         {
            setChildIndex(DisplayObject(this.hpmBoxes[7]),Math.min(xpAt,numChildren - 1));
            xpLine = this.hpmLineBox(this.mcExperienceBar,this);
            if(xpLine)
            {
               this.hpmBoxC[7] = new flash.geom.Point(xpLine.x + xpLine.width / 2,xpLine.y + xpLine.height / 2);
            }
         }
         // the widget's box with every part at rest: the corner the minimap movie places the widget by
         this.hpmRest = null;
         i = 0;
         while(i < numChildren)
         {
            this.hpmRest = this.hpmShownLocal(getChildAt(i),12,this.hpmRest);
            i++;
         }
         this.hpmApplyParts();
         i = 0;
         while(i < 4)
         {
            this.hpmSet(i,this.hpmValues[i]);
            i++;
         }
      }

      private function hpmFillBox(bar:DisplayObject) : Rectangle
      {
         var fill:DisplayObject = null;
         try
         {
            fill = Object(bar)["mcBar"] as DisplayObject;
         }
         catch(err:Error)
         {
            fill = null;
         }
         return (!fill ? bar : fill).getBounds(this);
      }

      // The experience bar's LINE in space's units. The bar (a CLIK StatusIndicator) shows its value by moving a mask
      // across its 101 frames: at a low value the mask reaches about 250 units left of the line, under the medallion
      // and past the screen's edge. The bar's own bounds follow that mask, so a box, a centre or a slider range taken
      // from them moved with the experience the player had. The line is the bar's flat pieces: of its children as wide
      // as half the widest, the thinnest and those up to twice as thick (the masks are several times thicker).
      private function hpmLineBox(bar:DisplayObject, space:DisplayObject) : Rectangle
      {
         var c:DisplayObjectContainer = bar as DisplayObjectContainer;
         var all:Array = [];
         var b:Rectangle = null;
         var u:Rectangle = null;
         var maxW:Number = 0;
         var minH:Number = Number.MAX_VALUE;
         var i:int = 0;
         if(!bar || !space)
         {
            return null;
         }
         if(!c || c.numChildren == 0)
         {
            return bar.getBounds(space);
         }
         while(i < c.numChildren)
         {
            b = c.getChildAt(i).getBounds(space);
            if(b.width > 0 && b.height > 0)
            {
               all.push(b);
               maxW = Math.max(maxW,b.width);
            }
            i++;
         }
         i = 0;
         while(i < all.length)
         {
            if(Rectangle(all[i]).width >= maxW * 0.5)
            {
               minH = Math.min(minH,Rectangle(all[i]).height);
            }
            i++;
         }
         i = 0;
         while(i < all.length)
         {
            b = Rectangle(all[i]);
            if(b.width >= maxW * 0.5 && b.height <= minH * 2 + 0.5)
            {
               u = !u ? b : u.union(b);
            }
            i++;
         }
         return !u ? bar.getBounds(space) : u;
      }

      // The experience line lined up with the health bar (the owner, 2026-10-13: "The level experience bar above the
      // health bar is out of position"): it runs from l to r, its bottom at bottom - the health bar's shown left edge and
      // right end, as far above the health bar as the stamina bar is below it. The game's own thickness is kept. The
      // level-up mark keeps its place beside the line's right end, as in the game.
      private function hpmPlaceXp(l:Number, r:Number, bottom:Number) : void
      {
         var bar:DisplayObject = this.mcExperienceBar;
         var mark:DisplayObject = this.mcNewLevelIndcator;
         var line:Rectangle = null;
         var mb:Rectangle = null;
         var dx:Number = 0;
         var dy:Number = 0;
         var m:Matrix = null;
         if(!bar || !(r > l))
         {
            return;
         }
         line = this.hpmLineBox(bar,this);
         if(!line || line.width <= 0)
         {
            return;
         }
         if(mark)
         {
            mb = mark.getBounds(this);
            dx = mb.left - line.right;
            dy = mb.top - line.bottom;
         }
         // through the matrix: the bar is a CLIK UIComponent, whose scaleX reads its size over its first size (1 at start,
         // not the timeline's 0.984) and whose scaleX setter resizes it as a component
         m = bar.transform.matrix;
         m.a *= (r - l) / line.width;
         bar.transform.matrix = m;
         line = this.hpmLineBox(bar,this);
         m = bar.transform.matrix;
         m.tx += l - line.left;
         m.ty += bottom - line.bottom;
         bar.transform.matrix = m;
         if(mark && mb)
         {
            line = this.hpmLineBox(bar,this);
            mark.x += line.right + dx - mb.left;
            mark.y += line.bottom + dy - mb.top;
         }
      }

      // moves every object in list by the step that puts key's centre at (x, cy) - mode 1: its right edge at x, mode 2: its
      // left edge at x
      private function hpmPlaceBadge(key:DisplayObject, list:Array, x:Number, cy:Number, mode:int = 0) : void
      {
         var s:Rectangle = null;
         var dx:Number = NaN;
         var dy:Number = NaN;
         var i:int = 0;
         if(!key)
         {
            return;
         }
         s = key.getBounds(this);
         if(mode == 1)
         {
            x -= s.width / 2;
         }
         else if(mode == 2)
         {
            x += s.width / 2;
         }
         dx = x - (s.left + s.width / 2);
         dy = cy - (s.top + s.height / 2);
         i = 0;
         while(i < list.length)
         {
            if(list[i])
            {
               DisplayObject(list[i]).x += dx;
               DisplayObject(list[i]).y += dy;
            }
            i++;
         }
         // the badges draw over the bars' left ends
         i = 0;
         while(i < list.length)
         {
            if(list[i] && DisplayObject(list[i]).parent == this)
            {
               setChildIndex(DisplayObject(list[i]),numChildren - 1);
            }
            i++;
         }
      }

      private function hpmHideRound() : void
      {
         var list:Array = [this.mcStaminaBar,this.mcToxicityBar,this.mcLockedToxicityBar,this.mcAdrenalinePoints];
         var i:int = 0;
         while(i < list.length)
         {
            if(list[i])
            {
               DisplayObject(list[i]).alpha = 0;
            }
            i++;
         }
      }

      // value 0..100
      public function hpmSet(which:int, value:Number) : void
      {
         this.hpmValues[which] = value;
         if(this.hpmFills.length > which)
         {
            Shape(this.hpmFills[which]).scaleX = Math.max(0,Math.min(100,value)) / 100;
         }
      }

      // Ciri has no toxicity or adrenaline: those two bars go with the game's own
      private var hpmIsCiri:Boolean = false;

      public function hpmCiri(isCiri:Boolean) : void
      {
         this.hpmIsCiri = isCiri;
         this.hpmApplyParts();
         this.hpmHideRound();
      }

      // ==============================================================================================================
      // The player widget broken up into its parts (the owner, 2026-10-07: "the player widget is a combined widget which
      // should be broken up into its individual parts, not as a single widget that can be added to the combined widget").
      // Each part is an element of the layout on its own: the medallion, the health bar, the experience (level) bar, the
      // stamina, adrenaline and toxicity bars, the sign badge and the toxicity skull. Each takes a move (percent of the screen), a size (about its
      // own centre), a length and height (the bars), Hide, Show (always / only in combat / only out of combat), Always
      // visible, Move with another widget or part, and the Combined widgets group - on top of where hpmBarsInit puts it,
      // inside the widget, which the minimap movie starts in the bottom left corner. The bars are our own sprites; each
      // other part is the game's own pieces moved into a container of ours (hpmWrap), so the game keeps showing and
      // hiding the pieces inside (the level-up mark, the sign text, Ciri's missing skull) and we move the whole.
      //
      // Always visible, per part: the game fades the WHOLE widget out a few seconds after its last change. While any shown
      // part is set to stay, the widget is not let fade (ShowElementFromState below) - the parts not set to stay are hidden
      // instead, and shown again as soon as the game shows the widget. When none is set, the game fades it as always.
      // HUDPositionManager.ws hands the whole layout string in through hpmSetLayout (the same string as the minimap's)
      // and combat through hpmSetCombat, and reads hpmReport (each part's box, for the page's slider ranges).
      // ==============================================================================================================
      private static const HPM_PART_KEYS:Array = ["Stamina","Adrenaline","Toxicity","Sign","Skull","Medallion","Health","Experience","Magic"];
      private var hpmRowBase:Array = [];      // [x, y] per bar, from hpmBarsInit
      private var hpmBoxes:Array = [null,null,null,null,null,null,null,null,null];   // the containers of parts 3..7
      private var hpmBoxC:Array = [null,null,null,null,null,null,null,null,null];    // each container's centre at rest (this widget's units)
      private var hpmParts:Array = [null,null,null,null,null,null,null,null,null];
      private var hpmOff:Object = {};          // every element's move and group membership, by key ("Move with")
      private var hpmPartsOn:Boolean = true;   // "Apply my layout"
      private var hpmAllAlways:Boolean = false;   // the page's "Always visible" for the whole HUD
      private var hpmGroupX:Number = 0;
      private var hpmGroupY:Number = 0;
      private var hpmWant:Boolean = false;     // what the game asked for last: the widget shown (true) or faded (false)
      private var hpmInCombat:Boolean = false;
      private var hpmCombatUntil:int = 0;      // out of combat counts once 3 s have passed (as in the minimap movie)
      public var hpmReport:String = "";

      // The widget's box AT REST (the owner, 2026-10-07: "whenever I try to push [the experience bar] left, it pulls the
      // character widget right"). The minimap movie placed the widget by its shown box's bottom left corner, and that box
      // grows with every part moved past it - so pushing a part left moved the whole widget right. It now places the
      // widget by this box, measured once with every part where hpmBarsInit puts it, so a part's move never moves the
      // others. (The effects' "Above the player widget" uses it too.)
      private var hpmRest:Rectangle = null;

      private function hpmShownLocal(root:DisplayObject, depth:int, box:Rectangle) : Rectangle
      {
         var i:int = 0;
         var c:DisplayObjectContainer = null;
         var b:Rectangle = null;
         if(!root || !root.visible || root.alpha <= 0 || depth < 0)
         {
            return box;
         }
         if(root == this.mcExperienceBar)
         {
            // by its line, not its value mask (hpmLineBox)
            b = this.hpmLineBox(root,this);
            return !b ? box : (!box ? b : box.union(b));
         }
         if(root is DisplayObjectContainer && DisplayObjectContainer(root).numChildren > 0 && depth > 0)
         {
            c = DisplayObjectContainer(root);
            while(i < c.numChildren)
            {
               box = this.hpmShownLocal(c.getChildAt(i),depth - 1,box);
               i++;
            }
            return box;
         }
         b = root.getBounds(this);
         if(b.width <= 0 || b.height <= 0)
         {
            return box;
         }
         return !box ? b : box.union(b);
      }

      // the rest box on the stage (read by the minimap movie)
      public function hpmRestBox() : Rectangle
      {
         var a:* = undefined;
         var b:* = undefined;
         if(!this.hpmRest || !stage)
         {
            return null;
         }
         a = this.localToGlobal(new flash.geom.Point(this.hpmRest.left,this.hpmRest.top));
         b = this.localToGlobal(new flash.geom.Point(this.hpmRest.right,this.hpmRest.bottom));
         return new Rectangle(a.x,a.y,b.x - a.x,b.y - a.y);
      }

      // moves the listed pieces (those that are this widget's own children) into a new container at the topmost one's depth,
      // keeping their order and their place on the screen
      private function hpmWrap(k:int, list:Array) : void
      {
         var box:Sprite = null;
         var found:Array = [];
         var top:int = -1;
         var i:int = 0;
         var d:DisplayObject = null;
         var b:Rectangle = null;
         while(i < list.length)
         {
            d = list[i] as DisplayObject;
            if(d && d.parent == this)
            {
               found.push({"d":d,"at":getChildIndex(d)});
               top = Math.max(top,getChildIndex(d));
            }
            i++;
         }
         if(top < 0)
         {
            return;
         }
         found.sortOn("at",Array.NUMERIC);
         box = new Sprite();
         box.mouseEnabled = false;
         addChildAt(box,top + 1);
         i = 0;
         while(i < found.length)
         {
            box.addChild(DisplayObject(found[i].d));
            i++;
         }
         b = box.getBounds(this);
         this.hpmBoxes[k] = box;
         this.hpmBoxC[k] = new flash.geom.Point(b.x + b.width / 2,b.y + b.height / 2);
      }

      public function hpmSetLayout(spec:String) : void
      {
         var parts:Array = !spec ? [] : spec.split("|");
         var i:int = 0;
         var f:Array = null;
         var k:int = 0;
         if(parts.length < 7 || parts[0] != "1")
         {
            return;
         }
         this.hpmPartsOn = parts[1] == "1";
         this.hpmAllAlways = parts[3] == "1";
         this.hpmGroupX = Number(parts[5]);
         this.hpmGroupY = Number(parts[6]);
         i = 7;
         while(i < parts.length)
         {
            f = String(parts[i]).split(",");
            i++;
            if(f.length < 13)
            {
               continue;
            }
            this.hpmOff[f[0]] = {"x":Number(f[1]),"y":Number(f[2]),"group":f[11] == "1"};
            k = HPM_PART_KEYS.indexOf(f[0]);
            if(k < 0)
            {
               continue;
            }
            this.hpmParts[k] = {"key":f[0],"x":Number(f[1]),"y":Number(f[2]),"s":Number(f[3]),"sx":Number(f[4]),"sy":Number(f[5]),
               "hide":f[6] == "1","show":int(f[7]),"always":f[8] == "1","with":f[10],"group":f[11] == "1"};
         }
         this.hpmApplyParts();
         this.hpmSyncHold();
      }

      public function hpmSetCombat(on:Boolean) : void
      {
         if(on)
         {
            this.hpmInCombat = true;
            this.hpmCombatUntil = 0;
            this.hpmSyncHold();
         }
         else if(this.hpmInCombat && this.hpmCombatUntil == 0)
         {
            this.hpmCombatUntil = flash.utils.getTimer() + 3000;
            setTimeout(this.hpmSyncHold,3100);
         }
      }

      private function hpmCombatNow() : Boolean
      {
         if(this.hpmInCombat && this.hpmCombatUntil > 0 && flash.utils.getTimer() >= this.hpmCombatUntil)
         {
            this.hpmInCombat = false;
            this.hpmCombatUntil = 0;
         }
         return this.hpmInCombat;
      }

      // the visible screen (Scaleform's visibleRect - the stage can reach past the screen's edges)
      private function hpmScreen() : Rectangle
      {
         var r:Rectangle = null;
         try
         {
            r = Object(stage).visibleRect as Rectangle;
         }
         catch(err:Error)
         {
            r = null;
         }
         if(!r || r.width <= 0)
         {
            r = new Rectangle(0,0,!stage ? 1920 : stage.stageWidth,!stage ? 1080 : stage.stageHeight);
         }
         return r;
      }

      // percent of the screen -> this widget's own units
      private function hpmLocal(xPct:Number, yPct:Number) : flash.geom.Point
      {
         var scr:Rectangle = this.hpmScreen();
         var a:* = this.globalToLocal(new flash.geom.Point(0,0));
         var b:* = this.globalToLocal(new flash.geom.Point(xPct / 100 * scr.width,yPct / 100 * scr.height));
         return new flash.geom.Point(b.x - a.x,b.y - a.y);
      }

      // a part's whole move: its own, the Combined widgets' if it is in the group, and the one of what it moves with
      private function hpmMove(o:Object) : flash.geom.Point
      {
         var w:Object = !o["with"] || o["with"] == o.key ? null : this.hpmOff[o["with"]];
         var x:Number = o.x + (o.group ? this.hpmGroupX : 0) + (!w ? 0 : w.x + (w.group ? this.hpmGroupX : 0));
         var y:Number = o.y + (o.group ? this.hpmGroupY : 0) + (!w ? 0 : w.y + (w.group ? this.hpmGroupY : 0));
         return this.hpmLocal(x,y);
      }

      private function hpmApplyParts() : void
      {
         var k:int = 0;
         var o:Object = null;
         var d:flash.geom.Point = null;
         var row:DisplayObject = null;
         var box:DisplayObject = null;
         var b:Rectangle = null;
         var c:flash.geom.Point = null;
         var sx:Number = NaN;
         var sy:Number = NaN;
         var ri:int = 0;
         while(k < HPM_PART_KEYS.length)
         {
            o = this.hpmPartsOn ? this.hpmParts[k] : null;
            sx = !o ? 1 : o.s * o.sx;
            sy = !o ? 1 : o.s * o.sy;
            d = !o ? new flash.geom.Point(0,0) : this.hpmMove(o);
            if(k < 3 || k == 8)
            {
               // the bars (Magic is the fourth row), about their own centre
               ri = k == 8 ? 3 : k;
               if(ri < this.hpmRows.length && ri < this.hpmRowBase.length)
               {
                  row = DisplayObject(this.hpmRows[ri]);
                  row.scaleX = 1;
                  row.scaleY = 1;
                  b = row.getBounds(row);
                  row.scaleX = sx;
                  row.scaleY = sy;
                  row.x = this.hpmRowBase[ri][0] + d.x - (b.x + b.width / 2) * (sx - 1);
                  row.y = this.hpmRowBase[ri][1] + d.y - (b.y + b.height / 2) * (sy - 1);
               }
            }
            else if(this.hpmBoxes[k])
            {
               // the other parts' containers, about the part's centre at rest
               box = DisplayObject(this.hpmBoxes[k]);
               c = this.hpmBoxC[k];
               box.scaleX = sx;
               box.scaleY = sy;
               box.x = d.x + c.x * (1 - sx);
               box.y = d.y + c.y * (1 - sy);
            }
            k++;
         }
         this.hpmApplyVis();
         this.hpmPlaceNumber();
         this.hpmPlaceMagicNumber();
      }

      // is part k set to stay shown when the game fades the widget
      private function hpmStays(k:int) : Boolean
      {
         var o:Object = this.hpmParts[k];
         if(k == 8 && !this.hpmMagicOn)
         {
            return false;
         }
         return this.hpmPartsOn && (this.hpmAllAlways || o && o.always) && this.hpmAllowed(k);
      }

      // Hide and Show (in / out of combat) allow part k now
      private function hpmAllowed(k:int) : Boolean
      {
         var o:Object = this.hpmPartsOn ? this.hpmParts[k] : null;
         var combat:Boolean = false;
         if(!o)
         {
            return true;
         }
         if(o.hide)
         {
            return false;
         }
         combat = this.hpmCombatNow();
         return o.show == 0 || o.show == 1 && combat || o.show == 2 && !combat;
      }

      // the widget is held up while any part allowed now is set to stay - never while the game has it hidden on purpose
      // (the HUD's player widget switched off: its state "Hide", or disabled)
      private function hpmHolding() : Boolean
      {
         var k:int = 0;
         if(!isEnabled || getState() == "Hide")
         {
            return false;
         }
         while(k < HPM_PART_KEYS.length)
         {
            if(this.hpmStays(k))
            {
               return true;
            }
            k++;
         }
         return false;
      }

      private function hpmApplyVis() : void
      {
         var held:Boolean = this.hpmHolding() && !this.hpmWant;
         var k:int = 0;
         var d:DisplayObject = null;
         while(k < HPM_PART_KEYS.length)
         {
            d = k < 3 ? (k < this.hpmRows.length ? DisplayObject(this.hpmRows[k]) : null) : (k == 8 ? (this.hpmRows.length > 3 ? DisplayObject(this.hpmRows[3]) : null) : DisplayObject(this.hpmBoxes[k]));
            if(d)
            {
               d.visible = this.hpmAllowed(k) && (!held || this.hpmStays(k)) && !(k < 3 && this.hpmIsCiri && k > 0) && (k != 8 || this.hpmMagicOn);
            }
            k++;
         }
         if(this.hpmFocusTf && this.hpmRows.length > 1)
         {
            this.hpmFocusTf.visible = this.hpmFocusNumOn && DisplayObject(this.hpmRows[1]).visible;
         }
      }

      // after a layout or combat change: hold the widget up, or hand it back to the game's own wish
      private function hpmSyncHold() : void
      {
         if(this.hpmHolding())
         {
            if(!_shown || alpha < 0.05)
            {
               super.ShowElementFromState(true,false);
            }
         }
         else if(!this.hpmWant && _shown)
         {
            super.ShowElementFromState(false,false);
         }
         this.hpmApplyVis();
      }

      // the game's fade in / out of the whole widget (its update timer, its display state)
      override public function ShowElementFromState(param1:Boolean, param2:Boolean = false) : void
      {
         this.hpmWant = param1;
         if(!param1 && this.hpmHolding())
         {
            super.ShowElementFromState(true,false);
         }
         else
         {
            super.ShowElementFromState(param1,param2);
         }
         this.hpmApplyVis();
      }

      // "<key>:<found>:<l>:<t>:<r>:<b>;..." for each part, in percent of the screen (the page's slider ranges)
      public function hpmMakeReport() : void
      {
         var s:String = "";
         var k:int = 0;
         var d:DisplayObject = null;
         var b:Rectangle = null;
         var scr:Rectangle = this.hpmScreen();
         while(k < HPM_PART_KEYS.length)
         {
            d = k < 3 ? (k < this.hpmRows.length ? DisplayObject(this.hpmRows[k]) : null) : (k == 8 ? (this.hpmRows.length > 3 ? DisplayObject(this.hpmRows[3]) : null) : DisplayObject(this.hpmBoxes[k]));
            if(d && d.stage && scr.width > 0 && scr.height > 0)
            {
               b = k != 7 ? d.getBounds(stage) : this.hpmXpBox(stage);
               s += HPM_PART_KEYS[k] + ":1:" + this.hpmPct(b.left - scr.left,scr.width) + ":" + this.hpmPct(b.top - scr.top,scr.height) + ":"
                  + this.hpmPct(b.right - scr.left,scr.width) + ":" + this.hpmPct(b.bottom - scr.top,scr.height) + ";";
            }
            else
            {
               s += HPM_PART_KEYS[k] + ":0;";
            }
            k++;
         }
         this.hpmReport = s;
      }

      // the Experience part's box: its line, and the level-up mark while the game shows it - never the bar's value mask,
      // which hung off the screen's left edge at a low value and left the page's left / right slider no room to the left
      private function hpmXpBox(space:DisplayObject) : Rectangle
      {
         var b:Rectangle = this.hpmLineBox(this.mcExperienceBar,space);
         if(b && this.mcNewLevelIndcator && this.mcNewLevelIndcator.visible)
         {
            b = b.union(this.mcNewLevelIndcator.getBounds(space));
         }
         return !b ? DisplayObject(this.hpmBoxes[7]).getBounds(space) : b;
      }

      private function hpmPct(v:Number, whole:Number) : String
      {
         return String(Math.round(v / whole * 1000) / 10);
      }
