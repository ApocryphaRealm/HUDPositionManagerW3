
      // ==============================================================================================================
      // Square Minimap (the owner, 2026-10-06): the game's own minimap, square, framed with its own ring art squared,
      // flush in the screen's top right corner. Inserted into HudModuleMinimap2 by tools/build_flash.py.
      // ==============================================================================================================

      private var sqmReady:Boolean = false;
      private var sqmRect:Rectangle;            // the round mask's box, in this module's space (the square's size)
      private var sqmFrameRect:Rectangle;       // the ring art's box (a little wider: its line shows round the map)
      private var sqmMaskCommon:Sprite;
      private var sqmMaskHub:Sprite;
      private var sqmMaskInterior:Sprite;
      private var sqmMaskShading:Sprite;
      private var sqmFrame:Sprite;
      private var sqmRing:DisplayObject;
      private var sqmNorthRotation:Number = NaN;
      private var sqmLastX:Number = NaN;
      private var sqmLastY:Number = NaN;
      private var sqmLastW:Number = NaN;

      // the ring art's colours (image 481 of the game's movie): a copper line and a dark plate lighter at the bottom
      private static const SQM_LINE_COLOR:uint = 0xE4B186;
      private static const SQM_LINE_ALPHA:Number = 0.72;
      private static const SQM_LINE_WIDTH:Number = 2;

      private function sqmSquare(rect:Rectangle) : Sprite
      {
         var s:Sprite = new Sprite();
         s.graphics.beginFill(0xFF00FF,1);
         s.graphics.drawRect(rect.x,rect.y,rect.width,rect.height);
         s.graphics.endFill();
         s.mouseEnabled = false;
         return s;
      }

      private function sqmFindRing() : DisplayObject
      {
         var i:int = 0;
         var d:DisplayObject = null;
         var b:Rectangle = null;
         // the ring is the one unnamed shape about 204 x 203 wide placed at the module's origin
         i = 0;
         while(i < numChildren)
         {
            d = getChildAt(i);
            if(d is Shape)
            {
               b = d.getBounds(this);
               if(b.width > 190 && b.width < 230 && b.height > 190 && b.height < 230)
               {
                  return d;
               }
            }
            i++;
         }
         return null;
      }

      private function sqmDrawFrame() : void
      {
         var g:* = this.sqmFrame.graphics;
         var r:Rectangle = this.sqmFrameRect;
         var m:Matrix = new Matrix();
         g.clear();
         // the plate: dark, lighter towards the bottom, as the ring art's disc
         m.createGradientBox(r.width,r.height,Math.PI / 2,r.x,r.y);
         g.beginGradientFill(GradientType.LINEAR,[0x000000,0x181818,0x8F8F8F],[1,1,1],[0,128,255],m);
         g.drawRect(r.x,r.y,r.width,r.height);
         g.endFill();
         // the copper line along the square's outer edge
         g.lineStyle(SQM_LINE_WIDTH,SQM_LINE_COLOR,SQM_LINE_ALPHA,true);
         g.drawRect(r.x + SQM_LINE_WIDTH / 2,r.y + SQM_LINE_WIDTH / 2,r.width - SQM_LINE_WIDTH,r.height - SQM_LINE_WIDTH);
      }

      public function sqmInit() : void
      {
         var c:Rectangle = null;
         var side:Number = NaN;
         if(this.sqmReady || !this.mcCommonMapMask)
         {
            return;
         }
         c = this.mcCommonMapMask.getBounds(this);
         side = Math.max(c.width,c.height);
         this.sqmRect = new Rectangle(c.x + c.width / 2 - side / 2,c.y + c.height / 2 - side / 2,side,side);
         this.sqmRing = this.sqmFindRing();
         if(this.sqmRing)
         {
            c = this.sqmRing.getBounds(this);
            side = Math.max(c.width,c.height);
            this.sqmFrameRect = new Rectangle(c.x + c.width / 2 - side / 2,c.y + c.height / 2 - side / 2,side,side);
         }
         else
         {
            this.sqmFrameRect = this.sqmRect.clone();
            this.sqmFrameRect.inflate(2,2);
         }
         this.sqmMaskCommon = this.sqmSquare(this.sqmRect);
         this.sqmMaskHub = this.sqmSquare(this.sqmRect);
         this.sqmMaskInterior = this.sqmSquare(this.sqmRect);
         this.sqmMaskShading = this.sqmSquare(this.sqmRect);
         addChild(this.sqmMaskCommon);
         addChild(this.sqmMaskHub);
         addChild(this.sqmMaskInterior);
         addChild(this.sqmMaskShading);
         this.sqmFrame = new Sprite();
         this.sqmFrame.mouseEnabled = false;
         this.sqmFrame.mouseChildren = false;
         this.sqmDrawFrame();
         if(this.sqmRing)
         {
            addChildAt(this.sqmFrame,getChildIndex(this.sqmRing));
            this.sqmRing.visible = false;
         }
         else
         {
            addChildAt(this.sqmFrame,0);
         }
         // the edge fade: the game's round shading clip, grown to cover the square's corners and cut square
         if(this.mcShading)
         {
            this.sqmGrowAboutCentre(this.mcShading,Math.SQRT2);
            this.mcShading.mask = this.sqmMaskShading;
         }
         this.sqmReady = true;
         this.sqmEnableMask(this.mcCommonMap.mask != null);
         addEventListener(Event.ENTER_FRAME,this.sqmOnFrame,false,0,true);
         this.sqmOnFrame(null);
      }

      private function sqmGrowAboutCentre(d:DisplayObject, k:Number) : void
      {
         var cx:Number = this.sqmRect.x + this.sqmRect.width / 2;
         var cy:Number = this.sqmRect.y + this.sqmRect.height / 2;
         d.x = cx + (d.x - cx) * k;
         d.y = cy + (d.y - cy) * k;
         d.scaleX *= k;
         d.scaleY *= k;
      }

      public function sqmEnableMask(on:Boolean) : Boolean
      {
         if(!this.sqmReady)
         {
            return false;
         }
         this.mcCommonMap.mask = !!on ? this.sqmMaskCommon : null;
         this.mcHubMap.mask = !!on ? this.sqmMaskHub : null;
         this.mcInteriorMap.mask = !!on ? this.sqmMaskInterior : null;
         return true;
      }

      // once a frame, cheap: keep the square flush with the screen's top right corner (the anchors can move the module on
      // a resolution or HUD scale change) and keep the north mark on the square's edge as the map turns
      private function sqmOnFrame(e:Event) : void
      {
         if(!stage || !parent)
         {
            return;
         }
         if(this.sqmLastX != this.x || this.sqmLastY != this.y || this.sqmLastW != stage.stageWidth)
         {
            this.sqmPlace();
         }
         if(this.mcNorthSign && this.mcNorthSign.rotation != this.sqmNorthRotation)
         {
            this.sqmPlaceNorth();
         }
         // the quest list: a few times a second (its content changes, the game's anchors can move it)
         this.sqmFrameCount++;
         if(this.sqmFrameCount >= 10)
         {
            this.sqmFrameCount = 0;
            this.sqmPlaceQuests();
         }
      }

      // the visible screen (the game's own helper: Scaleform's visibleRect - the stage can reach past the screen's edges,
      // which cut the frame off in the first build: the owner, 2026-10-06 "slightly out of frame")
      private function sqmScreen() : Rectangle
      {
         var r:Rectangle = CommonUtils.getScreenRect();
         if(!r || r.width <= 0)
         {
            r = new Rectangle(0,0,stage.stageWidth,stage.stageHeight);
         }
         return r;
      }

      private function sqmPlace() : void
      {
         var scr:Rectangle = this.sqmScreen();
         var corner:* = this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.right,this.sqmFrameRect.top));
         var target:* = new flash.geom.Point(scr.right - SQM_EDGE_RIGHT,scr.top + SQM_EDGE_TOP);
         var a:* = parent.globalToLocal(corner);
         var b:* = parent.globalToLocal(target);
         this.x += b.x - a.x;
         this.y += b.y - a.y;
         this.sqmLastX = this.x;
         this.sqmLastY = this.y;
         this.sqmLastW = stage.stageWidth;
         this.sqmPlaceQuests();
      }

      // the quest list directly underneath the square (the owner, 2026-10-06: "I want the quests area moved directly
      // underneath the minimap"). It is the game's own module "QuestsModule" in its own movie: found once by name in the
      // HUD's display tree, then moved so the top of what it shows sits a small gap below the square's frame.
      private var sqmFrameCount:int = 0;
      private var sqmQuests:DisplayObject;
      private var sqmQuestsSearched:int = 0;
      // the square's distance from the screen's top and right edges and the gap above the quest list, in the HUD's
      // units. The owner, 2026-10-06, after the first builds: "move the minimap out from both sides until the frame is
      // shown and the quest widget is clearly visible" - defaults 20 / 20 / 8, and a settings page to tune them
      // (SquareMinimap.ws passes them in through sqmSetSettings)
      private var SQM_EDGE_TOP:Number = 20;
      private var SQM_EDGE_RIGHT:Number = 20;
      private var SQM_QUEST_GAP:Number = 8;

      public function sqmSetSettings(top:Number, right:Number, gap:Number) : void
      {
         this.SQM_EDGE_TOP = Math.max(0,top);
         this.SQM_EDGE_RIGHT = Math.max(0,right);
         this.SQM_QUEST_GAP = Math.max(0,gap);
         if(this.sqmReady && stage && parent)
         {
            this.sqmPlace();
         }
      }

      private function sqmFindModule(root:DisplayObjectContainer, wanted:String, depth:int) : DisplayObject
      {
         var i:int = 0;
         var d:DisplayObject = null;
         var found:DisplayObject = null;
         if(!root || depth < 0)
         {
            return null;
         }
         i = 0;
         while(i < root.numChildren)
         {
            d = root.getChildAt(i);
            try
            {
               if(d.hasOwnProperty("moduleName") && Object(d).moduleName == wanted)
               {
                  return d;
               }
            }
            catch(err:Error)
            {
            }
            if(d is DisplayObjectContainer)
            {
               found = this.sqmFindModule(DisplayObjectContainer(d),wanted,depth - 1);
               if(found)
               {
                  return found;
               }
            }
            i++;
         }
         return null;
      }

      // the box of what the quest list SHOWS, in stage space: every visible, non-empty text line (TextField line metrics,
      // so a wide right- or centre-aligned field counts only its text). The module's own bounds start well above the
      // title and are wider than the text - the owner wants the text itself directly underneath, centred.
      private var sqmBoxL:Number;
      private var sqmBoxR:Number;
      private var sqmBoxT:Number;

      private function sqmTextBox(root:DisplayObjectContainer, depth:int) : void
      {
         var i:int = 0;
         var n:int = 0;
         var d:DisplayObject = null;
         var tf:TextField = null;
         var m:* = undefined;
         var y:Number = 2;
         var a:* = undefined;
         var b:* = undefined;
         if(!root || depth < 0 || !root.visible)
         {
            return;
         }
         i = 0;
         while(i < root.numChildren)
         {
            d = root.getChildAt(i);
            if(d.visible)
            {
               if(d is TextField)
               {
                  tf = TextField(d);
                  if(tf.text.length > 0)
                  {
                     y = 2;
                     n = 0;
                     while(n < tf.numLines)
                     {
                        m = tf.getLineMetrics(n);
                        if(m.width > 0)
                        {
                           a = tf.localToGlobal(new flash.geom.Point(m.x,y));
                           b = tf.localToGlobal(new flash.geom.Point(m.x + m.width,y));
                           if(isNaN(this.sqmBoxL) || a.x < this.sqmBoxL)
                           {
                              this.sqmBoxL = a.x;
                           }
                           if(isNaN(this.sqmBoxR) || b.x > this.sqmBoxR)
                           {
                              this.sqmBoxR = b.x;
                           }
                           if(isNaN(this.sqmBoxT) || a.y < this.sqmBoxT)
                           {
                              this.sqmBoxT = a.y;
                           }
                        }
                        y += m.height;
                        n++;
                     }
                  }
               }
               else if(d is DisplayObjectContainer)
               {
                  this.sqmTextBox(DisplayObjectContainer(d),depth - 1);
               }
            }
            i++;
         }
      }

      private function sqmPlaceQuests() : void
      {
         var bottom:* = undefined;
         var centre:* = undefined;
         var wantY:Number = NaN;
         var wantX:Number = NaN;
         var p:* = undefined;
         var q:* = undefined;
         if(!this.sqmReady || !stage)
         {
            return;
         }
         if(!this.sqmQuests || !this.sqmQuests.stage)
         {
            this.sqmQuests = null;
            if(this.sqmQuestsSearched > 40)
            {
               return;
            }
            this.sqmQuestsSearched++;
            this.sqmQuests = this.sqmFindModule(stage,"QuestsModule",6);
            if(!this.sqmQuests)
            {
               return;
            }
         }
         this.sqmBoxL = NaN;
         this.sqmBoxR = NaN;
         this.sqmBoxT = NaN;
         this.sqmTextBox(DisplayObjectContainer(this.sqmQuests),12);
         if(isNaN(this.sqmBoxT))
         {
            return;
         }
         // directly underneath the square, centred on it (the owner, 2026-10-06: "centered beneath the minimap")
         bottom = this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.x + this.sqmFrameRect.width / 2,this.sqmFrameRect.bottom));
         wantY = bottom.y + SQM_QUEST_GAP;
         wantX = bottom.x - (this.sqmBoxR - this.sqmBoxL) / 2;
         // never past the screen's right edge (a line wider than the square still shows whole)
         if(wantX + (this.sqmBoxR - this.sqmBoxL) > this.sqmScreen().right - SQM_EDGE_RIGHT)
         {
            wantX = this.sqmScreen().right - SQM_EDGE_RIGHT - (this.sqmBoxR - this.sqmBoxL);
         }
         if(Math.abs(this.sqmBoxT - wantY) < 0.5 && Math.abs(this.sqmBoxL - wantX) < 0.5)
         {
            return;
         }
         p = this.sqmQuests.parent.globalToLocal(new flash.geom.Point(this.sqmBoxL,this.sqmBoxT));
         q = this.sqmQuests.parent.globalToLocal(new flash.geom.Point(wantX,wantY));
         this.sqmQuests.x += q.x - p.x;
         this.sqmQuests.y += q.y - p.y;
      }

      // the north mark turns round the map's centre at the circle's radius; on a square its distance from the centre is
      // the half side over max(|cos|, |sin|) of its direction - the clip is scaled out by that much and its glyph back in
      public function sqmPlaceNorth() : void
      {
         var dir:Number = (this.mcNorthSign.rotation - 90) * Math.PI / 180;
         var k:Number = 1 / Math.max(Math.abs(Math.cos(dir)),Math.abs(Math.sin(dir)));
         var glyph:DisplayObject = null;
         this.mcNorthSign.scaleX = k;
         this.mcNorthSign.scaleY = k;
         if(this.mcNorthSign.numChildren > 0)
         {
            glyph = this.mcNorthSign.getChildAt(0);
            glyph.scaleX = 1 / k;
            glyph.scaleY = 1 / k;
         }
         this.sqmNorthRotation = this.mcNorthSign.rotation;
      }
