
      // ==============================================================================================================
      // HUD Position Manager (the owner, 2026-10-06): the game's own minimap, square, framed with its own ring art squared,
      // flush in the screen's top right corner. Inserted into HudModuleMinimap2 by tools/build_flash.py.
      // ==============================================================================================================

      private var sqmReady:Boolean = false;
      private var sqmRect:Rectangle;            // the round mask's box, in this module's space (the square's size)
      private var sqmFrameRect:Rectangle;       // the ring art's box (a little wider: its line shows round the map)
      private var sqmMaskCommon:Sprite;
      private var sqmMaskHub:Sprite;
      private var sqmMaskInterior:Sprite;
      private var sqmMaskShading:Sprite;
      private var sqmFrame:Sprite;
      private var sqmLine:Sprite;
      private var sqmRing:DisplayObject;
      private var sqmNorthRotation:Number = NaN;
      private var sqmLastX:Number = NaN;
      private var sqmLastY:Number = NaN;
      private var sqmLastW:Number = NaN;

      // the ring art's colours (image 481 of the game's movie): a copper line and a dark plate lighter at the bottom
      private static const SQM_LINE_COLOR:uint = 0xC99A72;
      private static const SQM_LINE_ALPHA:Number = 0.72;
      private static const SQM_LINE_WIDTH:Number = 2;

      private function sqmSquare(rect:Rectangle) : Sprite
      {
         var s:Sprite = new Sprite();
         s.graphics.beginFill(0xFF00FF,1);
         s.graphics.drawRect(rect.x,rect.y,rect.width,rect.height);
         s.graphics.endFill();
         s.mouseEnabled = false;
         return s;
      }

      private function sqmFindRing() : DisplayObject
      {
         var i:int = 0;
         var d:DisplayObject = null;
         var b:Rectangle = null;
         // the ring is the one unnamed shape about 204 x 203 wide placed at the module's origin
         i = 0;
         while(i < numChildren)
         {
            d = getChildAt(i);
            if(d is Shape)
            {
               b = d.getBounds(this);
               if(b.width > 190 && b.width < 230 && b.height > 190 && b.height < 230)
               {
                  return d;
               }
            }
            i++;
         }
         return null;
      }

      private function sqmDrawFrame() : void
      {
         var g:* = this.sqmFrame.graphics;
         var r:Rectangle = this.sqmRect;
         var m:Matrix = new Matrix();
         var lw:Number = SQM_LINE_WIDTH;
         g.clear();
         // the plate, only under the map (shows where a city map has no tile): dark, lighter towards the bottom
         m.createGradientBox(r.width,r.height,Math.PI / 2,r.x,r.y);
         g.beginGradientFill(GradientType.LINEAR,[0x000000,0x181818,0x8F8F8F],[1,1,1],[0,128,255],m);
         g.drawRect(r.x,r.y,r.width,r.height);
         g.endFill();
         // the copper line: four solid, opaque bands just outside the map, drawn ABOVE it - the same width and colour
         // on every side (the owner, 2026-10-06: the line was thicker and a different colour on two sides - it was a
         // see-through stroke over the plate's gradient, and the plate showed round the map unevenly)
         g = this.sqmLine.graphics;
         g.clear();
         g.beginFill(SQM_LINE_COLOR,1);
         g.drawRect(r.x - lw,r.y - lw,r.width + 2 * lw,lw);
         g.drawRect(r.x - lw,r.bottom,r.width + 2 * lw,lw);
         g.drawRect(r.x - lw,r.y,lw,r.height);
         g.drawRect(r.right,r.y,lw,r.height);
         g.endFill();
      }

      public function sqmInit() : void
      {
         var c:Rectangle = null;
         var side:Number = NaN;
         if(this.sqmReady || !this.mcCommonMapMask)
         {
            return;
         }
         c = this.mcCommonMapMask.getBounds(this);
         side = Math.max(c.width,c.height);
         this.sqmRect = new Rectangle(c.x + c.width / 2 - side / 2,c.y + c.height / 2 - side / 2,side,side);
         this.sqmRing = this.sqmFindRing();
         // the frame is the square grown by the line's width on every side, so the line shows evenly all round (the
         // ring art is not centred on the map: its line was thick on two sides and covered on the top and right)
         this.sqmFrameRect = this.sqmRect.clone();
         this.sqmFrameRect.inflate(SQM_LINE_WIDTH,SQM_LINE_WIDTH);   // the line's outer edge
         this.sqmMaskCommon = this.sqmSquare(this.sqmRect);
         this.sqmMaskHub = this.sqmSquare(this.sqmRect);
         this.sqmMaskInterior = this.sqmSquare(this.sqmRect);
         this.sqmMaskShading = this.sqmSquare(this.sqmRect);
         addChild(this.sqmMaskCommon);
         addChild(this.sqmMaskHub);
         addChild(this.sqmMaskInterior);
         addChild(this.sqmMaskShading);
         this.sqmFrame = new Sprite();
         this.sqmLine = new Sprite();
         this.sqmLine.mouseEnabled = false;
         this.sqmFrame.mouseEnabled = false;
         this.sqmFrame.mouseChildren = false;
         this.sqmDrawFrame();
         if(this.sqmRing)
         {
            addChildAt(this.sqmFrame,getChildIndex(this.sqmRing));
            this.sqmRing.visible = false;
         }
         else
         {
            addChildAt(this.sqmFrame,0);
         }
         addChild(this.sqmLine);
         // the edge fade: the game's round shading clip, grown to cover the square's corners and cut square
         if(this.mcShading)
         {
            this.sqmGrowAboutCentre(this.mcShading,Math.SQRT2);
            this.mcShading.mask = this.sqmMaskShading;
         }
         this.sqmReady = true;
         this.sqmEnableMask(this.mcCommonMap.mask != null);
         addEventListener(Event.ENTER_FRAME,this.sqmOnFrame,false,0,true);
         this.sqmOnFrame(null);
      }

      private function sqmGrowAboutCentre(d:DisplayObject, k:Number) : void
      {
         var cx:Number = this.sqmRect.x + this.sqmRect.width / 2;
         var cy:Number = this.sqmRect.y + this.sqmRect.height / 2;
         d.x = cx + (d.x - cx) * k;
         d.y = cy + (d.y - cy) * k;
         d.scaleX *= k;
         d.scaleY *= k;
      }

      public function sqmEnableMask(on:Boolean) : Boolean
      {
         if(!this.sqmReady)
         {
            return false;
         }
         this.mcCommonMap.mask = !!on ? this.sqmMaskCommon : null;
         this.mcHubMap.mask = !!on ? this.sqmMaskHub : null;
         this.mcInteriorMap.mask = !!on ? this.sqmMaskInterior : null;
         return true;
      }

      // =============================================================================================================
      // The layout (the owner, 2026-10-07: "HUD Position Manager completely reformatted and organized the way HUD
      // Position Manager is in Skyrim and Oblivion"). The page on the Apocrypha Menu Framework writes the whole layout as
      // one string; HUDPositionManager.ws hands it in through hpmSetLayout. Every element that is one of the game's HUD
      // modules is found by its moduleName in the HUD's display tree and given, on top of where the game puts it (or of
      // this mod's Place for it - the 1.0.0 spots): a move in percent of the screen, a size about its centre, hide, when
      // it shows (always / in combat / out of combat), always visible, move with another element and the Combined
      // widgets group. The player widget's parts are moved in hud_wolfstatbars and the dialogue choices in hud_dialog;
      // this movie reports whether the player widget is there for them.
      //
      // Cheap: an element left at its defaults with no Place is never touched. Each frame only the moved elements are
      // checked for a game move (two property reads); the boxes a Place needs are measured ten times fewer.
      // =============================================================================================================
      private var hpmEls:Array = [];            // the module elements, in the page's order
      private var hpmByKey:Object = {};
      private var hpmEnabled:Boolean = true;
      private var hpmAllVisible:Boolean = false;
      private var hpmGroupX:Number = 0;
      private var hpmGroupY:Number = 0;
      private var hpmLayoutSeen:Boolean = false;
      private var hpmInCombat:Boolean = false;
      private var hpmCombatUntil:int = 0;       // out of combat counts once 3 s have passed (no blinking between foes)
      public var sqmReport:String = "";         // read by HUDPositionManager.ws (GetMemberFlashString) for the page
      private var sqmFrameCount:int = 0;
      private var sqmQuestOver:Number = 0;      // how far the quest text centred under the square reaches past it

      // the module elements and the game's moduleName of each (Elements.h; the parts are not in this movie's table)
      private static const HPM_MODULES:Array = [["Minimap",""],["Quests","QuestsModule"],["Player","WolfHeadModule"],
         ["Buffs","BuffsModule"],["QuickItems","ItemInfoModule"],["Subtitles","SubtitlesModule"],["Messages","MessageModule"],
         ["Hints","ControlsFeedbackModule"],["Area","AreaInfoModule"],["Journal","JournalUpdateModule"],
         ["LootFeed","LootFeedModule"],["Enemy","EnemyFocusModule"],["Boss","BossFocusModule"],["Companion","CompanionModule"],
         ["Breath","OxygenBarModule"],["HorseStamina","HorseStaminaBarModule"],["HorseFear","HorsePanicBarModule"],
         ["Boat","BoatHealthModule"],["Damaged","DamagedItemsModule"],["TimeLeft","TimeLeftModule"],["TimeLapse","TimeLapseModule"]];
      private static const HPM_PARTS:Array = ["Stamina","Adrenaline","Toxicity","Sign","Skull","Dialogue"];

      private function hpmTable() : void
      {
         var i:int = 0;
         var o:Object = null;
         if(this.hpmEls.length > 0)
         {
            return;
         }
         while(i < HPM_MODULES.length)
         {
            o = {"key":HPM_MODULES[i][0],"module":HPM_MODULES[i][1],"m":null,"searched":0,
               "x":0,"y":0,"s":1,"sx":1,"sy":1,"hide":false,"show":0,"always":false,"place":false,"with":"","group":false,"zoom":100,
               "baseX":NaN,"baseY":NaN,"baseSX":NaN,"baseSY":NaN,"lastX":NaN,"lastY":NaN,"lastSX":NaN,"lastSY":NaN,
               "hiddenByUs":false,"active":false,"dirty":true};
            this.hpmEls.push(o);
            this.hpmByKey[o.key] = o;
            i++;
         }
         // the minimap is this module itself
         this.hpmByKey["Minimap"].m = this;
      }

      // "1|<enabled>|<unlocked>|<alwaysVisible>|<preview>|<groupX>|<groupY>|<key>,<x>,<y>,<scale>,<length>,<height>,<hide>,
      // <show>,<always>,<place>,<moveWith>,<groupMember>,<zoom>,<fill>,<grow>|..." (Settings.cpp LayoutString)
      public function hpmSetLayout(spec:String) : void
      {
         var parts:Array = !spec ? [] : spec.split("|");
         var i:int = 0;
         var f:Array = null;
         var o:Object = null;
         this.hpmTable();
         if(parts.length < 7 || parts[0] != "1")
         {
            return;
         }
         this.hpmEnabled = parts[1] == "1";
         this.hpmAllVisible = parts[3] == "1";
         this.hpmGroupX = Number(parts[5]);
         this.hpmGroupY = Number(parts[6]);
         i = 7;
         while(i < parts.length)
         {
            f = String(parts[i]).split(",");
            i++;
            if(f.length < 13)
            {
               continue;
            }
            o = this.hpmByKey[f[0]];
            if(!o)
            {
               continue;
            }
            o.x = Number(f[1]);
            o.y = Number(f[2]);
            o.s = Number(f[3]);
            o.sx = Number(f[4]);
            o.sy = Number(f[5]);
            o.hide = f[6] == "1";
            o.show = int(f[7]);
            o.always = f[8] == "1";
            o.place = f[9] == "1";
            o["with"] = f[10];
            o.group = f[11] == "1";
            o.zoom = Number(f[12]);
            o.dirty = true;   // re-applied on the next frame, from the game's place
         }
         this.hpmLayoutSeen = true;
         this.sqmQuestOver = 0;
         i = 0;
         while(i < this.hpmEls.length)
         {
            this.hpmActivate(this.hpmEls[i]);
            i++;
         }
      }

      // the zoom the layout asks for (HUDPositionManager.ws reads it for SetZoom)
      public function hpmZoom() : Number
      {
         var o:Object = !this.hpmByKey ? null : this.hpmByKey["Minimap"];
         return !o ? 100 : Number(o.zoom);
      }

      public function hpmSetCombat(on:Boolean) : void
      {
         if(on)
         {
            this.hpmInCombat = true;
            this.hpmCombatUntil = 0;
         }
         else if(this.hpmInCombat)
         {
            this.hpmCombatUntil = flash.utils.getTimer() + 3000;
         }
      }

      private function hpmCombatNow() : Boolean
      {
         if(this.hpmInCombat && this.hpmCombatUntil > 0 && flash.utils.getTimer() >= this.hpmCombatUntil)
         {
            this.hpmInCombat = false;
            this.hpmCombatUntil = 0;
         }
         return this.hpmInCombat;
      }

      // an element is "active" (looked at every frame) when anything about it differs from the game's own
      private function hpmActivate(o:Object) : void
      {
         var was:Boolean = o.active;
         o.active = this.hpmEnabled && (o.place || o.x != 0 || o.y != 0 || o.s != 1 || o.sx != 1 || o.sy != 1 || o.hide || o.show != 0
            || o.always || (this.hpmAllVisible && this.hpmFades(o.key)) || o["with"] != "" || (o.group && (this.hpmGroupX != 0 || this.hpmGroupY != 0)));
         if(was && !o.active)
         {
            this.hpmRelease(o);
         }
      }

      private function hpmFades(key:String) : Boolean
      {
         return key == "Player" || key == "QuickItems" || key == "Enemy" || key == "Boss" || key == "Companion" || key == "Breath"
            || key == "HorseStamina" || key == "HorseFear" || key == "Boat";
      }

      // back to the game's own place, size and visibility
      private function hpmRelease(o:Object) : void
      {
         var m:DisplayObject = o.m as DisplayObject;
         if(!m || !m.stage)
         {
            return;
         }
         if(!isNaN(o.baseX) && o.key != "Minimap")
         {
            m.x = o.baseX;
            m.y = o.baseY;
         }
         if(!isNaN(o.baseSX))
         {
            m.scaleX = o.baseSX;
            m.scaleY = o.baseSY;
         }
         if(o.hiddenByUs)
         {
            m.visible = true;
            o.hiddenByUs = false;
         }
         o.baseX = NaN;
         o.baseSX = NaN;
         o.lastX = NaN;
         o.lastSX = NaN;
      }

      private function hpmFind(o:Object) : DisplayObject
      {
         var m:DisplayObject = o.m as DisplayObject;
         if(m && m.stage)
         {
            return m;
         }
         o.m = null;
         if(o.searched > 40 || !stage)
         {
            return null;
         }
         o.searched++;
         m = this.sqmFindModule(stage,o.module,6);
         if(m)
         {
            o.m = m;
            o.baseX = NaN;
            o.baseSX = NaN;
            o.lastX = NaN;
            o.lastSX = NaN;
            o.dirty = true;
         }
         return m;
      }

      // percent of the screen -> stage units
      private function hpmPx(o:Object, scr:Rectangle) : Number
      {
         var w:Object = !o["with"] ? null : this.hpmByKey[o["with"]];
         var v:Number = o.x + (o.group ? this.hpmGroupX : 0) + (!w || w == o ? 0 : w.x + (w.group ? this.hpmGroupX : 0));
         return v / 100 * scr.width;
      }

      private function hpmPy(o:Object, scr:Rectangle) : Number
      {
         var w:Object = !o["with"] ? null : this.hpmByKey[o["with"]];
         var v:Number = o.y + (o.group ? this.hpmGroupY : 0) + (!w || w == o ? 0 : w.y + (w.group ? this.hpmGroupY : 0));
         return v / 100 * scr.height;
      }

      // stage units -> the module's parent units
      private function hpmToParent(m:DisplayObject, dx:Number, dy:Number) : flash.geom.Point
      {
         var a:* = m.parent.globalToLocal(new flash.geom.Point(0,0));
         var b:* = m.parent.globalToLocal(new flash.geom.Point(dx,dy));
         return new flash.geom.Point(b.x - a.x,b.y - a.y);
      }

      // once a frame, cheap: the square's north mark, the active elements (a game move puts them back on top of it)
      private function sqmOnFrame(e:Event) : void
      {
         var i:int = 0;
         var heavy:Boolean = false;
         if(!stage || !parent)
         {
            return;
         }
         if(this.mcNorthSign && this.mcNorthSign.rotation != this.sqmNorthRotation)
         {
            this.sqmPlaceNorth();
         }
         this.hpmTable();
         this.sqmFrameCount++;
         heavy = this.sqmFrameCount >= 10;
         if(heavy)
         {
            this.sqmFrameCount = 0;
         }
         if(!this.hpmLayoutSeen)
         {
            // no layout yet (the page's string arrives through the script): the 1.0.0 places, so the HUD never jumps
            this.hpmByKey["Minimap"].place = true;
            this.hpmByKey["Quests"].place = true;
            this.hpmByKey["Player"].place = true;
            this.hpmByKey["Buffs"].place = true;
            this.hpmByKey["QuickItems"].place = true;
            this.hpmLayoutSeen = true;
            i = 0;
            while(i < this.hpmEls.length)
            {
               this.hpmActivate(this.hpmEls[i]);
               i++;
            }
         }
         i = 0;
         while(i < this.hpmEls.length)
         {
            this.hpmApply(this.hpmEls[i],heavy);
            i++;
         }
         if(heavy)
         {
            this.hpmMakeReport();
         }
      }

      private function hpmApply(o:Object, heavy:Boolean) : void
      {
         var m:DisplayObject = null;
         var scr:Rectangle = null;
         var want:Boolean = false;
         var combat:Boolean = false;
         var k:Number = NaN;
         var d:flash.geom.Point = null;
         var gameMoved:Boolean = false;
         var resized:Boolean = false;
         if(!o.active)
         {
            if(heavy && !o.m)
            {
               this.hpmFind(o);   // for the report: is it in the HUD
            }
            return;
         }
         m = this.hpmFind(o);
         if(!m || !m.parent)
         {
            return;
         }
         // the game moved or resized it (an anchor update, a resolution or HUD scale change): its new place is the base.
         // lastX is where WE put it, so a layout change (dirty) re-applies from the same base instead of from our own move.
         gameMoved = isNaN(o.lastX) || m.x != o.lastX || m.y != o.lastY;
         if(gameMoved)
         {
            o.baseX = m.x;
            o.baseY = m.y;
         }
         resized = isNaN(o.lastSX) || m.scaleX != o.lastSX || m.scaleY != o.lastSY;
         if(resized)
         {
            o.baseSX = m.scaleX;
            o.baseSY = m.scaleY;
         }
         // visibility: Hide, Show in / out of combat, Always visible
         combat = this.hpmCombatNow();
         want = !o.hide && (o.show == 0 || o.show == 1 && combat || o.show == 2 && !combat);
         if(!want)
         {
            if(m.visible)
            {
               m.visible = false;
               o.hiddenByUs = true;
            }
         }
         else
         {
            if(o.hiddenByUs)
            {
               m.visible = true;
               o.hiddenByUs = false;
            }
            if((o.always || this.hpmAllVisible && this.hpmFades(o.key)) && m.alpha < 1)
            {
               m.alpha = 1;   // the module's own fade-out is overruled while this is on
            }
         }
         if(!gameMoved && !resized && !o.dirty && !heavy)
         {
            return;
         }
         o.dirty = false;
         scr = this.sqmScreen();
         // size about the element's own centre (its local box, measured when the size or the base changes)
         k = o.s;
         m.scaleX = o.baseSX * k * o.sx;
         m.scaleY = o.baseSY * k * o.sy;
         if(o.key == "Minimap" || o.key == "Quests" || o.key == "Player" || o.key == "Buffs" || o.key == "QuickItems")
         {
            if(o.place)
            {
               this.hpmPlace(o,m,scr);
               o.lastX = m.x;
               o.lastY = m.y;
               o.lastSX = m.scaleX;
               o.lastSY = m.scaleY;
               return;
            }
         }
         d = this.hpmToParent(m,this.hpmPx(o,scr),this.hpmPy(o,scr));
         m.x = o.baseX + d.x - this.hpmCentreShift(o,m,true);
         m.y = o.baseY + d.y - this.hpmCentreShift(o,m,false);
         o.lastX = m.x;
         o.lastY = m.y;
         o.lastSX = m.scaleX;
         o.lastSY = m.scaleY;
      }

      // how far its origin moves so the box keeps its centre when the size changes (in parent units)
      private function hpmCentreShift(o:Object, m:DisplayObject, horizontal:Boolean) : Number
      {
         var b:Rectangle = null;
         if(o.s * o.sx == 1 && o.s * o.sy == 1)
         {
            return 0;
         }
         b = m.getBounds(m);
         if(horizontal)
         {
            return (b.x + b.width / 2) * (m.scaleX - o.baseSX);
         }
         return (b.y + b.height / 2) * (m.scaleY - o.baseSY);
      }

      // this mod's spots (the 1.0.0 layout): a box edge of the element on a screen corner or beside another element, plus
      // the element's move (percent of the screen)
      private function hpmPlace(o:Object, m:DisplayObject, scr:Rectangle) : void
      {
         var box:Rectangle = null;
         var ox:Number = this.hpmPx(o,scr);
         var oy:Number = this.hpmPy(o,scr);
         var corner:* = undefined;
         var wolf:Object = null;
         var wolfBox:Rectangle = null;
         var mm:Object = null;
         var bottom:* = undefined;
         var over:Number = NaN;
         if(o.key == "Minimap")
         {
            if(!this.sqmReady)
            {
               return;
            }
            corner = this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.right,this.sqmFrameRect.top));
            this.hpmMoveTo(m,corner.x,corner.y,scr.right + ox - (this.hpmByKey["Quests"].place ? this.sqmQuestOver : 0),scr.top + oy);
         }
         else if(o.key == "Quests")
         {
            mm = this.hpmByKey["Minimap"];
            if(!this.sqmReady)
            {
               return;
            }
            this.sqmBoxL = NaN;
            this.sqmBoxR = NaN;
            this.sqmBoxT = NaN;
            this.sqmTextBox(DisplayObjectContainer(m),12);
            if(isNaN(this.sqmBoxT))
            {
               return;
            }
            bottom = this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.x + this.sqmFrameRect.width / 2,this.sqmFrameRect.bottom));
            // centred exactly under the square; text wider than the square moves the square in from the right (1.0.0)
            over = (this.sqmBoxR - this.sqmBoxL - (this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.right,0)).x - this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.left,0)).x)) / 2;
            if(mm.place && over > this.sqmQuestOver + 0.5)
            {
               this.sqmQuestOver = over;
               mm.dirty = true;
            }
            this.hpmMoveTo(m,this.sqmBoxL,this.sqmBoxT,bottom.x - (this.sqmBoxR - this.sqmBoxL) / 2 + ox,bottom.y + oy);
         }
         else if(o.key == "Player")
         {
            box = this.sqmShownBox(m,12,null);
            if(box)
            {
               this.hpmMoveTo(m,box.left,box.bottom,scr.left + ox,scr.bottom + oy);
            }
         }
         else if(o.key == "Buffs")
         {
            wolf = this.hpmByKey["Player"];
            wolfBox = !wolf.m ? null : this.sqmShownBox(DisplayObject(wolf.m),12,null);
            box = this.sqmShownBox(m,12,null);
            if(box && wolfBox)
            {
               this.hpmMoveTo(m,box.left,box.bottom,wolfBox.left + ox,wolfBox.top + oy);
            }
         }
         else if(o.key == "QuickItems")
         {
            box = this.sqmItemsBox(m);
            if(box)
            {
               this.hpmMoveTo(m,box.left,box.top,scr.left + ox,scr.top + oy);
            }
         }
      }

      // moves m so the stage point (fromX, fromY) of it lands on (toX, toY)
      private function hpmMoveTo(m:DisplayObject, fromX:Number, fromY:Number, toX:Number, toY:Number) : void
      {
         var p:* = undefined;
         var q:* = undefined;
         if(Math.abs(fromX - toX) < 0.5 && Math.abs(fromY - toY) < 0.5)
         {
            return;
         }
         p = m.parent.globalToLocal(new flash.geom.Point(fromX,fromY));
         q = m.parent.globalToLocal(new flash.geom.Point(toX,toY));
         m.x += q.x - p.x;
         m.y += q.y - p.y;
      }

      // "<key>:<found>:<l>:<t>:<r>:<b>;..." in percent of the screen, for the page's slider ranges (bridge.cpp)
      private function hpmMakeReport() : void
      {
         var s:String = "";
         var i:int = 0;
         var o:Object = null;
         var m:DisplayObject = null;
         var b:Rectangle = null;
         var scr:Rectangle = this.sqmScreen();
         var player:Boolean = false;
         while(i < this.hpmEls.length)
         {
            o = this.hpmEls[i];
            m = o.m as DisplayObject;
            if(m && m.stage)
            {
               b = o.key == "QuickItems" ? this.sqmItemsBox(m) : (o.key == "Minimap" ? this.hpmMinimapBox() : this.sqmShownBox(m,12,null));
               if(o.key == "Player")
               {
                  player = true;
               }
               if(b && b.width > 0)
               {
                  s += o.key + ":1:" + this.hpmPct(b.left - scr.left,scr.width) + ":" + this.hpmPct(b.top - scr.top,scr.height) + ":"
                     + this.hpmPct(b.right - scr.left,scr.width) + ":" + this.hpmPct(b.bottom - scr.top,scr.height) + ";";
               }
               else
               {
                  s += o.key + ":1;";
               }
            }
            else
            {
               s += o.key + ":0;";
            }
            i++;
         }
         i = 0;
         while(i < HPM_PARTS.length)
         {
            s += HPM_PARTS[i] + ":" + (HPM_PARTS[i] == "Dialogue" || player ? "1" : "0") + ";";
            i++;
         }
         this.sqmReport = s;
      }

      private function hpmPct(v:Number, whole:Number) : String
      {
         return whole <= 0 ? "0" : String(Math.round(v / whole * 1000) / 10);
      }

      private function hpmMinimapBox() : Rectangle
      {
         var a:* = this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.left,this.sqmFrameRect.top));
         var b:* = this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.right,this.sqmFrameRect.bottom));
         return new Rectangle(a.x,a.y,b.x - a.x,b.y - a.y);
      }

      // the visible screen (the game's own helper: Scaleform's visibleRect - the stage can reach past the screen's edges)
      private function sqmScreen() : Rectangle
      {
         var r:Rectangle = CommonUtils.getScreenRect();
         if(!r || r.width <= 0)
         {
            r = new Rectangle(0,0,stage.stageWidth,stage.stageHeight);
         }
         return r;
      }

      private function sqmFindModule(root:DisplayObjectContainer, wanted:String, depth:int) : DisplayObject
      {
         var i:int = 0;
         var d:DisplayObject = null;
         var found:DisplayObject = null;
         if(!root || depth < 0)
         {
            return null;
         }
         i = 0;
         while(i < root.numChildren)
         {
            d = root.getChildAt(i);
            try
            {
               if(d.hasOwnProperty("moduleName") && Object(d).moduleName == wanted)
               {
                  return d;
               }
            }
            catch(err:Error)
            {
            }
            if(d is DisplayObjectContainer)
            {
               found = this.sqmFindModule(DisplayObjectContainer(d),wanted,depth - 1);
               if(found)
               {
                  return found;
               }
            }
            i++;
         }
         return null;
      }

      // the box of what the quest list SHOWS, in stage space: every visible, non-empty text field (1.0.0)
      private var sqmBoxL:Number;
      private var sqmBoxR:Number;
      private var sqmBoxT:Number;

      private function sqmTextBox(root:DisplayObjectContainer, depth:int) : void
      {
         var i:int = 0;
         var n:int = 0;
         var d:DisplayObject = null;
         var tf:TextField = null;
         var m:* = undefined;
         var y:Number = 2;
         var a:* = undefined;
         var fb:Rectangle = null;
         if(!root || depth < 0 || !root.visible)
         {
            return;
         }
         i = 0;
         while(i < root.numChildren)
         {
            d = root.getChildAt(i);
            if(d.visible)
            {
               if(d is TextField)
               {
                  tf = TextField(d);
                  if(tf.text.length > 0)
                  {
                     fb = tf.getBounds(stage);
                     if(isNaN(this.sqmBoxL) || fb.left < this.sqmBoxL)
                     {
                        this.sqmBoxL = fb.left;
                     }
                     if(isNaN(this.sqmBoxR) || fb.right > this.sqmBoxR)
                     {
                        this.sqmBoxR = fb.right;
                     }
                     y = 2;
                     n = 0;
                     while(n < tf.numLines)
                     {
                        m = tf.getLineMetrics(n);
                        if(m.width > 0)
                        {
                           a = tf.localToGlobal(new flash.geom.Point(m.x,y));
                           if(isNaN(this.sqmBoxT) || a.y < this.sqmBoxT)
                           {
                              this.sqmBoxT = a.y;
                           }
                        }
                        y += m.height;
                        n++;
                     }
                  }
               }
               else if(d is DisplayObjectContainer)
               {
                  this.sqmTextBox(DisplayObjectContainer(d),depth - 1);
               }
            }
            i++;
         }
      }

      // the stage box of what a module SHOWS: the union of its visible, non-empty leaves (1.0.0)
      private function sqmShownBox(root:DisplayObject, depth:int, box:Rectangle) : Rectangle
      {
         var i:int = 0;
         var c:DisplayObjectContainer = null;
         var b:Rectangle = null;
         if(!root || !root.visible || root.alpha <= 0 || depth < 0)
         {
            return box;
         }
         if(root is DisplayObjectContainer && DisplayObjectContainer(root).numChildren > 0 && depth > 0)
         {
            c = DisplayObjectContainer(root);
            i = 0;
            while(i < c.numChildren)
            {
               box = this.sqmShownBox(c.getChildAt(i),depth - 1,box);
               i++;
            }
            return box;
         }
         b = root.getBounds(stage);
         if(b.width <= 0 || b.height <= 0)
         {
            return box;
         }
         return !box ? b : box.union(b);
      }

      // the quick items' box: every slot and hint, shown or not, so the slots never jump when a hint appears (1.0.0)
      private function sqmItemsBox(m:DisplayObject) : Rectangle
      {
         var names:Array = ["mcItemSlot","mcPotionSlot1","mcPotionSlot2","mcKbPotion1","mcKbPotion2","mcKbPotion3",
            "mcKbPotion4","mcKbOilSteel","mcKbOilSilver","btnSwitchHint","btnSwitchHintDoubleTap","btnSwitchHintOils",
            "btnSwitchHintOilsKB"];
         var box:Rectangle = null;
         var b:Rectangle = null;
         var d:DisplayObject = null;
         var i:int = 0;
         while(i < names.length)
         {
            d = null;
            try
            {
               d = Object(m)[names[i]] as DisplayObject;
            }
            catch(err:Error)
            {
               d = null;
            }
            if(d)
            {
               b = d.getBounds(stage);
               if(b.width > 0 && b.height > 0)
               {
                  box = !box ? b : box.union(b);
               }
            }
            i++;
         }
         return box;
      }

      // after a zoom change (the Zoom setting, or the game's own interior / boat zoom): the hub map only asks for the

      // the square dark until Geralt walked (2026-10-07, at 50 %). Ask for them now with the current position - the same

      }

      // the north mark turns round the map's centre at the circle's radius; on a square its distance from the centre is
      // the half side over max(|cos|, |sin|) of its direction - the clip is scaled out by that much and its glyph back in
      public function sqmPlaceNorth() : void
      {
         var dir:Number = (this.mcNorthSign.rotation - 90) * Math.PI / 180;
         var m:Number = Math.max(Math.abs(Math.cos(dir)),Math.abs(Math.sin(dir)));
         var glyph:DisplayObject = null;
         var gb:Rectangle = null;
         var cx:Number = NaN;
         var cy:Number = NaN;
         var d0:Number = 0;
         var r:Number = 0;
         var half:Number = !this.sqmRect ? 0 : this.sqmRect.width / 2;
         var k:Number = 1 / m;
         // just INSIDE the frame, not on it: on the edge half the mark stood out past the square, and at the top the
         // screen cut it off (open since 2026-10-06). Measured in this module's space with the clip at scale 1: d0 is
         // the glyph's distance from the clip's pivot (the map centre) - the game's ring radius, larger than the half
         // square - and r its own radius. Its centre goes to (half - r - line) / m from the centre on the square.
         this.mcNorthSign.scaleX = 1;
         this.mcNorthSign.scaleY = 1;
         // the mark is the clip's own drawing when it has no child (seen 2026-10-07: the child branch never ran)
         if(this.mcNorthSign.numChildren > 0)
         {
            glyph = this.mcNorthSign.getChildAt(0);
            if(isNaN(this.sqmGlyphX0))
            {
               this.sqmGlyphX0 = glyph.x;
               this.sqmGlyphY0 = glyph.y;
            }
            glyph.scaleX = 1;
            glyph.scaleY = 1;
            glyph.x = this.sqmGlyphX0;
            glyph.y = this.sqmGlyphY0;
         }
         gb = this.mcNorthSign.getBounds(this);
         if(gb.width > 0)
         {
            cx = gb.x + gb.width / 2 - this.mcNorthSign.x;
            cy = gb.y + gb.height / 2 - this.mcNorthSign.y;
            d0 = Math.sqrt(cx * cx + cy * cy);
            r = Math.max(gb.width,gb.height) / 2;
            if(d0 > 0 && half > r + SQM_LINE_WIDTH)
            {
               k = (half - r - SQM_LINE_WIDTH) / m / d0;
            }
         }
         this.mcNorthSign.scaleX = k;
         this.mcNorthSign.scaleY = k;
         if(glyph)
         {
            // back to its own size, about its own CENTRE: the glyph clip's origin is the map centre, not the mark (the
            // first try scaled about the origin and put the mark straight back on the edge - logged 2026-10-07)
            gb = glyph.getBounds(glyph);
            glyph.scaleX = 1 / k;
            glyph.scaleY = 1 / k;
            glyph.x = this.sqmGlyphX0 + (gb.x + gb.width / 2) * (1 - 1 / k);
            glyph.y = this.sqmGlyphY0 + (gb.y + gb.height / 2) * (1 - 1 / k);
         }
         this.sqmNorthRotation = this.mcNorthSign.rotation;
      }

      private var sqmGlyphX0:Number = NaN;
      private var sqmGlyphY0:Number = NaN;
