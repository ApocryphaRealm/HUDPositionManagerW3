
      // ==============================================================================================================
      // HUD Position Manager (the owner, 2026-10-06): the game's own minimap, square, framed with its own ring art squared,
      // flush in the screen's top right corner. Inserted into HudModuleMinimap2 by tools/build_flash.py.
      // ==============================================================================================================

      private var sqmReady:Boolean = false;
      private var sqmRect:Rectangle;            // the round mask's box, in this module's space (the square's size)
      private var sqmFrameRect:Rectangle;       // the ring art's box (a little wider: its line shows round the map)
      private var sqmMaskCommon:Sprite;
      private var sqmMaskHub:Sprite;
      private var sqmMaskInterior:Sprite;
      private var sqmMaskShading:Sprite;
      private var sqmFrame:Sprite;
      private var sqmLine:Sprite;
      private var sqmRing:DisplayObject;
      private var sqmNorthRotation:Number = NaN;
      private var sqmLastX:Number = NaN;
      private var sqmLastY:Number = NaN;
      private var sqmLastW:Number = NaN;

      // the ring art's colours (image 481 of the game's movie): a copper line and a dark plate lighter at the bottom
      private static const SQM_LINE_COLOR:uint = 0xC99A72;
      private static const SQM_LINE_ALPHA:Number = 0.72;
      private static const SQM_LINE_WIDTH:Number = 2;

      private function sqmSquare(rect:Rectangle) : Sprite
      {
         var s:Sprite = new Sprite();
         s.graphics.beginFill(0xFF00FF,1);
         s.graphics.drawRect(rect.x,rect.y,rect.width,rect.height);
         s.graphics.endFill();
         s.mouseEnabled = false;
         return s;
      }

      private function sqmFindRing() : DisplayObject
      {
         var i:int = 0;
         var d:DisplayObject = null;
         var b:Rectangle = null;
         // the ring is the one unnamed shape about 204 x 203 wide placed at the module's origin
         i = 0;
         while(i < numChildren)
         {
            d = getChildAt(i);
            if(d is Shape)
            {
               b = d.getBounds(this);
               if(b.width > 190 && b.width < 230 && b.height > 190 && b.height < 230)
               {
                  return d;
               }
            }
            i++;
         }
         return null;
      }

      private function sqmDrawFrame() : void
      {
         var g:* = this.sqmFrame.graphics;
         var r:Rectangle = this.sqmRect;
         var m:Matrix = new Matrix();
         var lw:Number = SQM_LINE_WIDTH;
         g.clear();
         // the plate, only under the map (shows where a city map has no tile): dark, lighter towards the bottom
         m.createGradientBox(r.width,r.height,Math.PI / 2,r.x,r.y);
         g.beginGradientFill(GradientType.LINEAR,[0x000000,0x181818,0x8F8F8F],[1,1,1],[0,128,255],m);
         g.drawRect(r.x,r.y,r.width,r.height);
         g.endFill();
         // the copper line: four solid, opaque bands just outside the map, drawn ABOVE it - the same width and colour
         // on every side (the owner, 2026-10-06: the line was thicker and a different colour on two sides - it was a
         // see-through stroke over the plate's gradient, and the plate showed round the map unevenly)
         g = this.sqmLine.graphics;
         g.clear();
         g.beginFill(SQM_LINE_COLOR,1);
         g.drawRect(r.x - lw,r.y - lw,r.width + 2 * lw,lw);
         g.drawRect(r.x - lw,r.bottom,r.width + 2 * lw,lw);
         g.drawRect(r.x - lw,r.y,lw,r.height);
         g.drawRect(r.right,r.y,lw,r.height);
         g.endFill();
      }

      public function sqmInit() : void
      {
         var c:Rectangle = null;
         var side:Number = NaN;
         if(this.sqmReady || !this.mcCommonMapMask)
         {
            return;
         }
         c = this.mcCommonMapMask.getBounds(this);
         side = Math.max(c.width,c.height);
         this.sqmRect = new Rectangle(c.x + c.width / 2 - side / 2,c.y + c.height / 2 - side / 2,side,side);
         this.sqmRing = this.sqmFindRing();
         // the frame is the square grown by the line's width on every side, so the line shows evenly all round (the
         // ring art is not centred on the map: its line was thick on two sides and covered on the top and right)
         this.sqmFrameRect = this.sqmRect.clone();
         this.sqmFrameRect.inflate(SQM_LINE_WIDTH,SQM_LINE_WIDTH);   // the line's outer edge
         this.sqmMaskCommon = this.sqmSquare(this.sqmRect);
         this.sqmMaskHub = this.sqmSquare(this.sqmRect);
         this.sqmMaskInterior = this.sqmSquare(this.sqmRect);
         this.sqmMaskShading = this.sqmSquare(this.sqmRect);
         addChild(this.sqmMaskCommon);
         addChild(this.sqmMaskHub);
         addChild(this.sqmMaskInterior);
         addChild(this.sqmMaskShading);
         this.sqmFrame = new Sprite();
         this.sqmLine = new Sprite();
         this.sqmLine.mouseEnabled = false;
         this.sqmFrame.mouseEnabled = false;
         this.sqmFrame.mouseChildren = false;
         this.sqmDrawFrame();
         if(this.sqmRing)
         {
            addChildAt(this.sqmFrame,getChildIndex(this.sqmRing));
            this.sqmRing.visible = false;
         }
         else
         {
            addChildAt(this.sqmFrame,0);
         }
         addChild(this.sqmLine);
         // the edge fade: the game's round shading clip, grown to cover the square's corners and cut square
         if(this.mcShading)
         {
            this.sqmGrowAboutCentre(this.mcShading,Math.SQRT2);
            this.mcShading.mask = this.sqmMaskShading;
         }
         this.sqmReady = true;
         this.sqmEnableMask(this.mcCommonMap.mask != null);
         addEventListener(Event.ENTER_FRAME,this.sqmOnFrame,false,0,true);
         this.sqmOnFrame(null);
      }

      private function sqmGrowAboutCentre(d:DisplayObject, k:Number) : void
      {
         var cx:Number = this.sqmRect.x + this.sqmRect.width / 2;
         var cy:Number = this.sqmRect.y + this.sqmRect.height / 2;
         d.x = cx + (d.x - cx) * k;
         d.y = cy + (d.y - cy) * k;
         d.scaleX *= k;
         d.scaleY *= k;
      }

      public function sqmEnableMask(on:Boolean) : Boolean
      {
         if(!this.sqmReady)
         {
            return false;
         }
         this.mcCommonMap.mask = !!on ? this.sqmMaskCommon : null;
         this.mcHubMap.mask = !!on ? this.sqmMaskHub : null;
         this.mcInteriorMap.mask = !!on ? this.sqmMaskInterior : null;
         return true;
      }

      // once a frame, cheap: keep the square flush with the screen's top right corner (the anchors can move the module on
      // a resolution or HUD scale change) and keep the north mark on the square's edge as the map turns
      private function sqmOnFrame(e:Event) : void
      {
         if(!stage || !parent)
         {
            return;
         }
         if(this.sqmLastX != this.x || this.sqmLastY != this.y || this.sqmLastW != stage.stageWidth)
         {
            this.sqmPlace();
         }
         if(this.mcNorthSign && this.mcNorthSign.rotation != this.sqmNorthRotation)
         {
            this.sqmPlaceNorth();
         }
         // the quest list: a few times a second (its content changes, the game's anchors can move it)
         this.sqmFrameCount++;
         if(this.sqmFrameCount >= 10)
         {
            this.sqmFrameCount = 0;
            this.sqmPlaceQuests();
            this.sqmPlacePlayer();
         }
      }

      // the visible screen (the game's own helper: Scaleform's visibleRect - the stage can reach past the screen's edges,
      // which cut the frame off in the first build: the owner, 2026-10-06 "slightly out of frame")
      private function sqmScreen() : Rectangle
      {
         var r:Rectangle = CommonUtils.getScreenRect();
         if(!r || r.width <= 0)
         {
            r = new Rectangle(0,0,stage.stageWidth,stage.stageHeight);
         }
         return r;
      }

      private function sqmPlace() : void
      {
         var scr:Rectangle = this.sqmScreen();
         var corner:* = this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.right,this.sqmFrameRect.top));
         var target:* = new flash.geom.Point(scr.right - SQM_EDGE_RIGHT - this.sqmQuestOver,scr.top + SQM_EDGE_TOP);
         var a:* = parent.globalToLocal(corner);
         var b:* = parent.globalToLocal(target);
         this.x += b.x - a.x;
         this.y += b.y - a.y;
         this.sqmLastX = this.x;
         this.sqmLastY = this.y;
         this.sqmLastW = stage.stageWidth;
         if(!this.sqmPlacing)
         {
            this.sqmPlacing = true;
            this.sqmPlaceQuests();
            this.sqmPlacing = false;
         }
      }

      // the quest list directly underneath the square (the owner, 2026-10-06: "I want the quests area moved directly
      // underneath the minimap"). It is the game's own module "QuestsModule" in its own movie: found once by name in the
      // HUD's display tree, then moved so the top of what it shows sits a small gap below the square's frame.
      private var sqmFrameCount:int = 0;
      private var sqmQuests:DisplayObject;
      private var sqmQuestsSearched:int = 0;
      // the square's distance from the screen's top and right edges and the gap above the quest list, in the HUD's
      // units. The owner, 2026-10-06, after the first builds: "move the minimap out from both sides until the frame is
      // shown and the quest widget is clearly visible" - defaults 20 / 20 / 8, and a settings page to tune them
      // (HUDPositionManager.ws passes them in through sqmSetSettings)
      private var SQM_EDGE_TOP:Number = 4;
      private var SQM_EDGE_RIGHT:Number = 4;
      private var SQM_QUEST_GAP:Number = 8;
      // how far the centred quest text reaches past each side of the square (the widest seen this session - it only
      // grows, so the square does not jump back and forth as quests change); the square moves in by that much
      private var sqmQuestOver:Number = 0;
      private var sqmPlacing:Boolean = false;

      public function sqmSetSettings(top:Number, right:Number, gap:Number) : void
      {
         this.SQM_EDGE_TOP = Math.max(0,top);
         this.SQM_EDGE_RIGHT = Math.max(0,right);
         this.SQM_QUEST_GAP = Math.max(0,gap);
         if(this.sqmReady && stage && parent)
         {
            this.sqmPlace();
         }
      }

      private function sqmFindModule(root:DisplayObjectContainer, wanted:String, depth:int) : DisplayObject
      {
         var i:int = 0;
         var d:DisplayObject = null;
         var found:DisplayObject = null;
         if(!root || depth < 0)
         {
            return null;
         }
         i = 0;
         while(i < root.numChildren)
         {
            d = root.getChildAt(i);
            try
            {
               if(d.hasOwnProperty("moduleName") && Object(d).moduleName == wanted)
               {
                  return d;
               }
            }
            catch(err:Error)
            {
            }
            if(d is DisplayObjectContainer)
            {
               found = this.sqmFindModule(DisplayObjectContainer(d),wanted,depth - 1);
               if(found)
               {
                  return found;
               }
            }
            i++;
         }
         return null;
      }

      // the box of what the quest list SHOWS, in stage space: every visible, non-empty text line (TextField line metrics,
      // so a wide right- or centre-aligned field counts only its text). The module's own bounds start well above the
      // title and are wider than the text - the owner wants the text itself directly underneath, centred.
      private var sqmBoxL:Number;
      private var sqmBoxR:Number;
      private var sqmBoxT:Number;

      private function sqmTextBox(root:DisplayObjectContainer, depth:int) : void
      {
         var i:int = 0;
         var n:int = 0;
         var d:DisplayObject = null;
         var tf:TextField = null;
         var m:* = undefined;
         var y:Number = 2;
         var a:* = undefined;
         var fb:Rectangle = null;
         var b:* = undefined;
         if(!root || depth < 0 || !root.visible)
         {
            return;
         }
         i = 0;
         while(i < root.numChildren)
         {
            d = root.getChildAt(i);
            if(d.visible)
            {
               if(d is TextField)
               {
                  tf = TextField(d);
                  if(tf.text.length > 0)
                  {
                     // across: the text box itself - the game wraps its text inside it, so no line reaches past it
                     // (line metrics under-measured a long objective line, which then ran off the screen)
                     fb = tf.getBounds(stage);
                     if(isNaN(this.sqmBoxL) || fb.left < this.sqmBoxL)
                     {
                        this.sqmBoxL = fb.left;
                     }
                     if(isNaN(this.sqmBoxR) || fb.right > this.sqmBoxR)
                     {
                        this.sqmBoxR = fb.right;
                     }
                     y = 2;
                     n = 0;
                     while(n < tf.numLines)
                     {
                        m = tf.getLineMetrics(n);
                        if(m.width > 0)
                        {
                           a = tf.localToGlobal(new flash.geom.Point(m.x,y));
                           if(isNaN(this.sqmBoxT) || a.y < this.sqmBoxT)
                           {
                              this.sqmBoxT = a.y;
                           }
                        }
                        y += m.height;
                        n++;
                     }
                  }
               }
               else if(d is DisplayObjectContainer)
               {
                  this.sqmTextBox(DisplayObjectContainer(d),depth - 1);
               }
            }
            i++;
         }
      }

      private function sqmPlaceQuests() : void
      {
         var bottom:* = undefined;
         var over:Number = NaN;
         var wantY:Number = NaN;
         var wantX:Number = NaN;
         var p:* = undefined;
         var q:* = undefined;
         if(!this.sqmReady || !stage)
         {
            return;
         }
         if(!this.sqmQuests || !this.sqmQuests.stage)
         {
            this.sqmQuests = null;
            if(this.sqmQuestsSearched > 40)
            {
               return;
            }
            this.sqmQuestsSearched++;
            this.sqmQuests = this.sqmFindModule(stage,"QuestsModule",6);
            if(!this.sqmQuests)
            {
               return;
            }
         }
         this.sqmBoxL = NaN;
         this.sqmBoxR = NaN;
         this.sqmBoxT = NaN;
         this.sqmTextBox(DisplayObjectContainer(this.sqmQuests),12);
         if(isNaN(this.sqmBoxT))
         {
            return;
         }
         // directly underneath the square, centred on it (the owner, 2026-10-06: "centered beneath the minimap")
         bottom = this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.x + this.sqmFrameRect.width / 2,this.sqmFrameRect.bottom));
         wantY = bottom.y + SQM_QUEST_GAP;
         wantX = bottom.x - (this.sqmBoxR - this.sqmBoxL) / 2;
         // centred exactly; if the text is wider than the square, the square itself moves in from the right edge far
         // enough for the whole text to show (the owner, 2026-10-06: "move the minimap to where it's far enough away
         // from the side so that the quest widget can properly display the words while it's centered")
         over = (this.sqmBoxR - this.sqmBoxL - (this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.right,0)).x - this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.left,0)).x)) / 2;
         if(over > this.sqmQuestOver + 0.5)
         {
            this.sqmQuestOver = over;
            this.sqmPlacing = true;
            this.sqmPlace();
            this.sqmPlacing = false;
            bottom = this.localToGlobal(new flash.geom.Point(this.sqmFrameRect.x + this.sqmFrameRect.width / 2,this.sqmFrameRect.bottom));
            wantY = bottom.y + SQM_QUEST_GAP;
            wantX = bottom.x - (this.sqmBoxR - this.sqmBoxL) / 2;
         }
         if(Math.abs(this.sqmBoxT - wantY) < 0.5 && Math.abs(this.sqmBoxL - wantX) < 0.5)
         {
            return;
         }
         p = this.sqmQuests.parent.globalToLocal(new flash.geom.Point(this.sqmBoxL,this.sqmBoxT));
         q = this.sqmQuests.parent.globalToLocal(new flash.geom.Point(wantX,wantY));
         this.sqmQuests.x += q.x - p.x;
         this.sqmQuests.y += q.y - p.y;
      }

      // after a zoom change (the Zoom setting, or the game's own interior / boat zoom): the hub map only asks for the
      // tiles around the player when the player MOVES past a threshold (SetPlayerPosition), so a new zoom left part of
      // the square dark until Geralt walked (2026-10-07, at 50 %). Ask for them now with the current position - the same
      // call SetPlayerPosition makes - once the position and the world are known.
      public function sqmRefreshTiles() : void
      {
         if(m_worldSize <= 0 || m_playerTileX < 0 || isNaN(m_playerTexturePosX) || !this.mcHubMap)
         {
            return;
         }
         this.mcHubMap.mcHubMapContainer.UpdatePosition(m_radius,m_playerTexturePosX,m_playerTexturePosY);
      }

      // the player widget (the game's "WolfHeadModule": medallion, vitality, stamina, toxicity, adrenaline) in the
      // screen's bottom left instead of the top left (the owner, 2026-10-07: "the player's HUD widget in the top left. I
      // want it in the bottom left"), PLAYER_LEFT / PLAYER_BOTTOM HUD units in from the edges; its buff icons
      // ("BuffsModule", normally just underneath it) sit just above it, left aligned. Re-applied a few times a second,
      // as for the quest list: the game's anchors put the modules back on a HUD scale or resolution change.
      private var PLAYER_LEFT:Number = 20;
      private var PLAYER_BOTTOM:Number = 20;
      private var sqmWolf:DisplayObject;
      private var sqmBuffs:DisplayObject;
      private var sqmWolfSearched:int = 0;
      private var sqmBuffsSearched:int = 0;

      public function sqmSetPlayer(left:Number, bottom:Number) : void
      {
         this.PLAYER_LEFT = Math.max(0,left);
         this.PLAYER_BOTTOM = Math.max(0,bottom);
         if(this.sqmReady && stage)
         {
            this.sqmPlacePlayer();
         }
      }

      // the stage box of what a module SHOWS: the union of its visible, non-empty leaves (a module's own bounds include
      // hidden parts - the wolf head carries bars and icons that are off most of the time)
      private function sqmShownBox(root:DisplayObject, depth:int, box:Rectangle) : Rectangle
      {
         var i:int = 0;
         var c:DisplayObjectContainer = null;
         var b:Rectangle = null;
         if(!root || !root.visible || root.alpha <= 0 || depth < 0)
         {
            return box;
         }
         if(root is DisplayObjectContainer && DisplayObjectContainer(root).numChildren > 0 && depth > 0)
         {
            c = DisplayObjectContainer(root);
            i = 0;
            while(i < c.numChildren)
            {
               box = this.sqmShownBox(c.getChildAt(i),depth - 1,box);
               i++;
            }
            return box;
         }
         b = root.getBounds(stage);
         if(b.width <= 0 || b.height <= 0)
         {
            return box;
         }
         return !box ? b : box.union(b);
      }

      private function sqmMoveBox(d:DisplayObject, box:Rectangle, wantLeft:Number, wantBottom:Number) : void
      {
         var p:* = undefined;
         var q:* = undefined;
         if(Math.abs(box.left - wantLeft) < 0.5 && Math.abs(box.bottom - wantBottom) < 0.5)
         {
            return;
         }
         p = d.parent.globalToLocal(new flash.geom.Point(box.left,box.bottom));
         q = d.parent.globalToLocal(new flash.geom.Point(wantLeft,wantBottom));
         d.x += q.x - p.x;
         d.y += q.y - p.y;
      }

      private function sqmPlacePlayer() : void
      {
         var scr:Rectangle = null;
         var wolfBox:Rectangle = null;
         var buffsBox:Rectangle = null;
         if(!this.sqmReady || !stage)
         {
            return;
         }
         if(!this.sqmWolf || !this.sqmWolf.stage)
         {
            this.sqmWolf = null;
            if(this.sqmWolfSearched > 40)
            {
               return;
            }
            this.sqmWolfSearched++;
            this.sqmWolf = this.sqmFindModule(stage,"WolfHeadModule",6);
            if(!this.sqmWolf)
            {
               return;
            }
         }
         scr = this.sqmScreen();
         wolfBox = this.sqmShownBox(this.sqmWolf,12,null);
         if(!wolfBox)
         {
            return;
         }
         this.sqmMoveBox(this.sqmWolf,wolfBox,scr.left + PLAYER_LEFT,scr.bottom - PLAYER_BOTTOM);
         wolfBox = this.sqmShownBox(this.sqmWolf,12,null);
         if(!this.sqmBuffs || !this.sqmBuffs.stage)
         {
            this.sqmBuffs = null;
            if(this.sqmBuffsSearched <= 40)
            {
               this.sqmBuffsSearched++;
               this.sqmBuffs = this.sqmFindModule(stage,"BuffsModule",6);
            }
         }
         if(this.sqmBuffs && wolfBox)
         {
            buffsBox = this.sqmShownBox(this.sqmBuffs,12,null);
            if(buffsBox)
            {
               this.sqmMoveBox(this.sqmBuffs,buffsBox,wolfBox.left,wolfBox.top - 6);
            }
         }
      }

      // the north mark turns round the map's centre at the circle's radius; on a square its distance from the centre is
      // the half side over max(|cos|, |sin|) of its direction - the clip is scaled out by that much and its glyph back in
      public function sqmPlaceNorth() : void
      {
         var dir:Number = (this.mcNorthSign.rotation - 90) * Math.PI / 180;
         var m:Number = Math.max(Math.abs(Math.cos(dir)),Math.abs(Math.sin(dir)));
         var glyph:DisplayObject = null;
         var gb:Rectangle = null;
         var cx:Number = NaN;
         var cy:Number = NaN;
         var d0:Number = 0;
         var r:Number = 0;
         var half:Number = !this.sqmRect ? 0 : this.sqmRect.width / 2;
         var k:Number = 1 / m;
         // just INSIDE the frame, not on it: on the edge half the mark stood out past the square, and at the top the
         // screen cut it off (open since 2026-10-06). Measured in this module's space with the clip at scale 1: d0 is
         // the glyph's distance from the clip's pivot (the map centre) - the game's ring radius, larger than the half
         // square - and r its own radius. Its centre goes to (half - r - line) / m from the centre on the square.
         this.mcNorthSign.scaleX = 1;
         this.mcNorthSign.scaleY = 1;
         // the mark is the clip's own drawing when it has no child (seen 2026-10-07: the child branch never ran)
         if(this.mcNorthSign.numChildren > 0)
         {
            glyph = this.mcNorthSign.getChildAt(0);
            if(isNaN(this.sqmGlyphX0))
            {
               this.sqmGlyphX0 = glyph.x;
               this.sqmGlyphY0 = glyph.y;
            }
            glyph.scaleX = 1;
            glyph.scaleY = 1;
            glyph.x = this.sqmGlyphX0;
            glyph.y = this.sqmGlyphY0;
         }
         gb = this.mcNorthSign.getBounds(this);
         if(gb.width > 0)
         {
            cx = gb.x + gb.width / 2 - this.mcNorthSign.x;
            cy = gb.y + gb.height / 2 - this.mcNorthSign.y;
            d0 = Math.sqrt(cx * cx + cy * cy);
            r = Math.max(gb.width,gb.height) / 2;
            if(d0 > 0 && half > r + SQM_LINE_WIDTH)
            {
               k = (half - r - SQM_LINE_WIDTH) / m / d0;
            }
         }
         this.mcNorthSign.scaleX = k;
         this.mcNorthSign.scaleY = k;
         if(glyph)
         {
            // back to its own size, about its own CENTRE: the glyph clip's origin is the map centre, not the mark (the
            // first try scaled about the origin and put the mark straight back on the edge - logged 2026-10-07)
            gb = glyph.getBounds(glyph);
            glyph.scaleX = 1 / k;
            glyph.scaleY = 1 / k;
            glyph.x = this.sqmGlyphX0 + (gb.x + gb.width / 2) * (1 - 1 / k);
            glyph.y = this.sqmGlyphY0 + (gb.y + gb.height / 2) * (1 - 1 / k);
         }
         this.sqmNorthRotation = this.mcNorthSign.rotation;
      }

      private var sqmGlyphX0:Number = NaN;
      private var sqmGlyphY0:Number = NaN;
