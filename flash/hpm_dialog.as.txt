
      // =============================================================================================================
      // HUD Position Manager - the dialogue choices as an element of the layout (the owner, 2026-10-07: "move the
      // dialogue selections to the right of the screen so they don't block the view of the person you're talking to";
      // then the page organised as the Skyrim and Oblivion versions are). The game draws the choices in
      // mcOptionContainer, right of centre, with lines up to 540 units wide. Its Place "At the right edge" moves the whole
      // container sideways so the end of its longest line meets the screen's right edge; then the element's move (percent
      // of the screen) and size apply, and Hide hides the list (the game still takes the choice from the keys). Placed
      // when the game sets the choices and re-checked a few times a second while they show (a controller switch re-renders
      // the lines without their number prefix). HUDPositionManager.ws hands the layout string in through hpmSetLayout.
      // =============================================================================================================
      private var hpmDlgOn:Boolean = true;      // "Apply my layout"
      private var hpmDlgPlace:Boolean = true;
      private var hpmDlgX:Number = 0;
      private var hpmDlgY:Number = 0;
      private var hpmDlgS:Number = 1;
      private var hpmDlgHide:Boolean = false;
      private var hpmChoiceX0:Number = NaN;
      private var hpmChoiceY0:Number = NaN;
      private var hpmChoiceS0:Number = NaN;
      private var hpmChoiceFrames:int = 0;
      private var hpmChoiceListening:Boolean = false;

      public function hpmSetLayout(spec:String) : void
      {
         var parts:Array = !spec ? [] : spec.split("|");
         var i:int = 0;
         var f:Array = null;
         if(parts.length < 7 || parts[0] != "1")
         {
            return;
         }
         this.hpmDlgOn = parts[1] == "1";
         i = 7;
         while(i < parts.length)
         {
            f = String(parts[i]).split(",");
            i++;
            if(f[0] != "Dialogue" || f.length < 10)
            {
               continue;
            }
            this.hpmDlgX = Number(f[1]);
            this.hpmDlgY = Number(f[2]);
            this.hpmDlgS = Number(f[3]);
            this.hpmDlgHide = f[6] == "1";
            this.hpmDlgPlace = f[9] == "1";
         }
         if(!this.hpmDlgPlace)
         {
            this.hpmDlgReport = "";
         }
         this.hpmPlaceChoices();
      }

      public function hpmPlaceChoices() : void
      {
         var scr:Rectangle = null;
         var c:OptionContainer = this.mcOptionContainer;
         var n:int = 0;
         var i:int = 0;
         var opt:Option = null;
         var right:Number = NaN;
         var r:Number = NaN;
         var a:* = undefined;
         var b:* = undefined;
         var on:Boolean = false;
         var free:Number = NaN;
         if(!c || !c.parent || !stage)
         {
            return;
         }
         if(isNaN(this.hpmChoiceX0))
         {
            this.hpmChoiceX0 = c.x;
            this.hpmChoiceY0 = c.y;
            this.hpmChoiceS0 = c.scaleX;
         }
         if(!this.hpmChoiceListening)
         {
            this.hpmChoiceListening = true;
            addEventListener(Event.ENTER_FRAME,this.hpmChoiceFrame,false,0,true);
         }
         scr = CommonUtils.getScreenRect();
         if(!scr || scr.width <= 0)
         {
            scr = new Rectangle(0,0,stage.stageWidth,stage.stageHeight);
         }
         on = this.hpmDlgOn;
         // size, from the game's own; the list's own origin is where its lines start, so the size grows it to the right
         c.scaleX = this.hpmChoiceS0 * (on ? this.hpmDlgS : 1);
         c.scaleY = this.hpmChoiceS0 * (on ? this.hpmDlgS : 1);
         c.alpha = on && this.hpmDlgHide ? 0 : c.alpha == 0 && this.hpmChoiceHidden ? 1 : c.alpha;
         this.hpmChoiceHidden = on && this.hpmDlgHide;
         // the move, percent of the screen in the container's parent's units
         a = c.parent.globalToLocal(new flash.geom.Point(0,0));
         b = c.parent.globalToLocal(new flash.geom.Point((on ? this.hpmDlgX : 0) / 100 * scr.width,(on ? this.hpmDlgY : 0) / 100 * scr.height));
         c.y = this.hpmChoiceY0 + (b.y - a.y);
         free = this.hpmChoiceX0 + (b.x - a.x);   // where the move alone puts it (no anchor)
         if(!on || !this.hpmDlgPlace)
         {
            c.x = free;
            return;
         }
         if(!c.mcOptionList || !c.mcOptionList.dataProvider)
         {
            return;
         }
         n = Math.min(c.mcOptionList.dataProvider.length,7);
         i = 0;
         while(i < n)
         {
            opt = c.mcOptionList.getRendererAt(i) as Option;
            if(opt && opt.tfLine && opt.tfLine.text != "" && opt.tfLine.textWidth > 0)
            {
               r = opt.tfLine.localToGlobal(new flash.geom.Point(opt.tfLine.textWidth + 4,0)).x;
               if(isNaN(right) || r > right)
               {
                  right = r;
               }
            }
            i++;
         }
         if(isNaN(right))
         {
            return;
         }
         // the longest line's end on the screen's right edge, then the move
         if(Math.abs(scr.right + this.hpmDlgX / 100 * scr.width - right) >= 0.5)
         {
            a = c.parent.globalToLocal(new flash.geom.Point(right,0));
            b = c.parent.globalToLocal(new flash.geom.Point(scr.right + this.hpmDlgX / 100 * scr.width,0));
            c.x += b.x - a.x;
         }
         // The anchor is gone from the page (the owner, 2026-10-07: "Get rid of the anchors"): while an older settings file
         // still has it on, report the move that keeps the list where the anchor put it, for the bridge to write into the
         // sliders (HUDPositionManager.ws reads hpmDlgReport). Measured with the lines of the conversation on screen.
         a = c.parent.localToGlobal(new flash.geom.Point(free,0));
         b = c.parent.localToGlobal(new flash.geom.Point(c.x,0));
         if(scr.width > 0)
         {
            this.hpmDlgReport = "Anchor:Dialogue:" + String(Math.round((this.hpmDlgX + (b.x - a.x) / scr.width * 100) * 100) / 100) + ":"
               + String(Math.round(this.hpmDlgY * 100) / 100) + ";";
         }
      }

      public var hpmDlgReport:String = "";

      private var hpmChoiceHidden:Boolean = false;

      // cheap: only while the choices show, every 10th frame
      private function hpmChoiceFrame(e:Event) : void
      {
         if(!this.mcOptionContainer || !this.mcOptionContainer.visible)
         {
            return;
         }
         this.hpmChoiceFrames++;
         if(this.hpmChoiceFrames >= 10)
         {
            this.hpmChoiceFrames = 0;
            this.hpmPlaceChoices();
         }
      }
